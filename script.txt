local function isWhitelisted(player)
    if not player then return false end
    local nameLower = string.lower(player.Name)
    local dispLower = string.lower(player.DisplayName)
    for k, _ in pairs(WHITELISTED_USERS) do
        local kLower = string.lower(k)
        if nameLower == kLower or dispLower == kLower then
            return true
        end
    end
    return false
end

local function findPlayer(nameQuery)
    if not nameQuery then return nil end
    local query = string.lower(nameQuery)
    local allP = Players:GetPlayers()
    for i = 1, #allP do
        local p = allP[i]
        if string.lower(p.Name) == query or string.lower(p.DisplayName) == query or string.find(string.lower(p.Name), query) or string.find(string.lower(p.DisplayName), query) then
            return p
        end
    end
    return nil
end

local function collectLocalCharacterModels()
    local list = {}
    local seen = {}
    local function add(model)
        if model and model:FindFirstChild("HumanoidRootPart") and not seen[model] then
            seen[model] = true
            list[#list + 1] = model
        end
    end
    add(getHoodCharacterModel(LocalPlayer))
    add(LocalPlayer.Character)
    return list
end

local function getBodyEffects(char, player)
    if char then
        local be = char:FindFirstChild("BodyEffects")
        if be then return be end
    end
    if not player and char then
        player = Players:GetPlayerFromCharacter(char)
    end
    if player then
        local hoodModel = getHoodCharacterModel(player)
        if hoodModel then
            return hoodModel:FindFirstChild("BodyEffects")
        end
    end
    return nil
end

local function isCharacterKnocked(char, player)
    if not char and player then

        player = Players:GetPlayerFromCharacter(char)

    local hum = char:FindFirstChildOfClass("Humanoid")

    if be:FindFirstChild("K.O") and be["K.O"].Value == true then return true end

end

local function isHumKnockLikeState(hum)
    if not hum then return false end
    local state = hum:GetState()
    return state == Enum.HumanoidStateType.FallingDown
        or state == Enum.HumanoidStateType.Ragdoll
        or state == Enum.HumanoidStateType.Physics

local function isBodyEffectsKnocked(be, hum)
    if be then
        local ko = be:FindFirstChild("K.O")
        if ko and ko:IsA("BoolValue") and ko.Value == true then return true end
        local dead = be:FindFirstChild("Dead")
        if dead and dead:IsA("BoolValue") and dead.Value == true then return true end
        local beHealth = be:FindFirstChild("Health")
        if beHealth and beHealth:IsA("NumberValue") and beHealth.Value <= 0 then return true end
        local beHealthClient = be:FindFirstChild("Health_CLIENT")
        if beHealthClient and beHealthClient:IsA("NumberValue") and beHealthClient.Value <= 0 then return true end
    end
    if hum then
        if hum.Health <= 0 then return true end
        if hum.PlatformStand then return true end
        if isHumKnockLikeState(hum) then return true end
        if hum.Health > 0 and hum.Health <= 5 then return true end
    end
    return false
end

local function getLocalAntiStompCharacter()
    local models = collectLocalCharacterModels()
    for i = 1, #models do
        local c = models[i]
        local hum = c:FindFirstChildOfClass("Humanoid")
        local be = c:FindFirstChild("BodyEffects") or getBodyEffects(c, LocalPlayer)
        if isBodyEffectsKnocked(be, hum) then
            return c, c:FindFirstChild("HumanoidRootPart"), hum, be
        end
    end
    for i = 1, #models do
        local c = models[i]
        local hum = c:FindFirstChildOfClass("Humanoid")
        if hum and (hum.PlatformStand or isHumKnockLikeState(hum)) then
            local be = c:FindFirstChild("BodyEffects") or getBodyEffects(c, LocalPlayer)
            return c, c:FindFirstChild("HumanoidRootPart"), hum, be
        end
    end
    return nil, nil, nil, nil
end

local function isStompAnimationTrack(track)
    if not track then return false end
    if string.find(string.lower(track.Name), "stomp") then return true end
    local anim = track.Animation
    if not anim then return false end
    if hoodStompAnimation and anim == hoodStompAnimation then return true end
    if string.lower(anim.Name) == "stomp" then return true end
    if STOMP_ANIM_IDS[anim.AnimationId] then return true end
    return false
end

local function isPlayerPerformingStomp(player, myPos)
    local pChar = player and getHoodCharacterModel(player)
    if not pChar then return false end
    local pHum = pChar:FindFirstChildOfClass("Humanoid")
    local pHrp = pChar:FindFirstChild("HumanoidRootPart")
    if not pHum or not pHrp then return false end

    if (pHrp.Position - myPos).Magnitude > 10 then return false end

    for _, track in ipairs(pHum:GetPlayingAnimationTracks()) do
        if isStompAnimationTrack(track) then return true end
    end

    local pBe = getBodyEffects(pChar, player)
    if pBe then
        local attacking = pBe:FindFirstChild("Attacking")
        if attacking and attacking:IsA("BoolValue") and attacking.Value == true then
            if pHrp.Position.Y > myPos.Y + 0.5 then return true end
        end
    end

    if pHrp.Position.Y > myPos.Y + 0.8 and (Vector3.new(pHrp.Position.X, 0, pHrp.Position.Z) - Vector3.new(myPos.X, 0, myPos.Z)).Magnitude <= 4.5 then
        return true
    end
    return false
end

local ANTI_STOMP_NEAR_RANGE = 14

local function isNearbyStompThreat(char, myHrp)
    if not char or not myHrp then return false end
    local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or myHrp
    local myPos = torso.Position
    local players = Players:GetPlayers()
    for i = 1, #players do
        local p = players[i]
        if p ~= LocalPlayer then
            local pChar = getHoodCharacterModel(p)
            local pHrp = pChar and pChar:FindFirstChild("HumanoidRootPart")
            if pHrp then
                local dist = (pHrp.Position - myPos).Magnitude
                if dist <= ANTI_STOMP_NEAR_RANGE then
                    if isPlayerPerformingStomp(p, myPos) then return true end
                    local flat = (Vector3.new(pHrp.Position.X, 0, pHrp.Position.Z) - Vector3.new(myPos.X, 0, myPos.Z)).Magnitude
                    if flat <= 9 then return true end
                    if pHrp.Position.Y > myPos.Y + 0.15 then return true end
                end
            end
        end
    end
    return false
end

            end
            local ko = be:FindFirstChild("K.O")

            if beHealth and beHealth:IsA("NumberValue") then beHealth.Value = 0 end
        end)
    end
    if myHum then
        pcall(function() myHum.Health = 0 end)

local function runAntiStompTick()
    if not isAntiStompActive then return end

    local char, myHrp, myHum = getLocalAntiStompCharacter()
    if not char or not myHrp or not myHum then
        resetAntiStompSession()
        return
    end

    if antiStompFlickerRestore ~= nil then
        myHrp.AssemblyLinearVelocity = antiStompFlickerRestore
        antiStompFlickerRestore = nil
    end

    local stompThreat = isNearbyStompThreat(char, myHrp)
    if not stompThreat then
        antiStompThreatSince = nil
        if tick() - antiStompKoSyncTick >= 0.35 then
            antiStompKoSyncTick = tick()
            fireMainEventRequestKOUpdate()
        end
        return
    end

    if not antiStompThreatSince then
        antiStompThreatSince = tick()
    end

    fireMainEventRequestKOUpdate()
    applyAntiStompVelocityFlicker(myHrp)
    pcall(function()
        myHrp.CFrame = myHrp.CFrame + Vector3.new(0, 9, 0)
        myHrp.AssemblyLinearVelocity = Vector3.zero
        myHrp.AssemblyAngularVelocity = Vector3.zero
    end)
    syncAllLocalRootsTo(myHrp)
    forceAntiStompRespawn(char, myHum)
end

    if currentDanceTrack then
        pcall(function()

    if danceAnimObj then

local function playSyncDance(danceData, targetTimePos)
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end

    if currentDanceTrack then
        pcall(function() currentDanceTrack:Stop(); currentDanceTrack:Destroy() end)
        currentDanceTrack = nil
    end

    local anim = Instance.new("Animation")
    anim.AnimationId = danceData.Id
    danceAnimObj = anim

    local track = hum:LoadAnimation(anim)
    track.Priority = Enum.AnimationPriority.Action
    track.Looped = true
    track:Play()
    if targetTimePos and track.Length > 0 then
        track.TimePosition = targetTimePos % track.Length
    end
    currentDanceTrack = track

local function getBestSilentTarget()
    if not SilentAim.Enabled then return nil end

    if currentTarget and currentTarget.Character and not isWhitelisted(currentTarget) then
        local char = currentTarget.Character
        local hum = char:FindFirstChildOfClass("Humanoid")
        local head = char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")

        if hum and hum.Health > 0 and head then
            local be = char:FindFirstChild("BodyEffects")
            local isKO = be and (
                (be:FindFirstChild("K.O") and be["K.O"].Value == true) or
                (be:FindFirstChild("Dead") and be.Dead.Value == true)
            )
            if not isKO then
                return { Part = head, Character = char }
            end
        end
    end

    local bestTarget = nil
    local shortestDist = SilentAim.FOV
    local mousePos = UserInputService:GetMouseLocation()
    local localHrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not localHrp then return nil end

    local players = Players:GetPlayers()
    for i = 1, #players do
        local player = players[i]
        if player ~= LocalPlayer and not isWhitelisted(player) and player.Character then
            local char = player.Character
            local hum = char:FindFirstChildOfClass("Humanoid")
            local head = char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")

            if hum and hum.Health > 0 and head then
                local worldDist = (head.Position - localHrp.Position).Magnitude
                if worldDist <= SilentAim.MaxDistance then
                    local be = char:FindFirstChild("BodyEffects")
                    local isKO = be and (
                        (be:FindFirstChild("K.O") and be["K.O"].Value == true) or
                        (be:FindFirstChild("Dead") and be.Dead.Value == true)
                    )

                    if not isKO then
                        local screenPos, onScreen = Camera:WorldToViewportPoint(head.Position)
                        if onScreen then
                            local screen2D = Vector2.new(screenPos.X, screenPos.Y)
                            local dist = (screen2D - mousePos).Magnitude
                            if dist <= shortestDist then
                                shortestDist = dist
                                bestTarget = { Part = head, Character = char }
                            end
                        end
                    end
                end
            end
        end
    end
    return bestTarget
end

local function applyGunHandlerHook()
    if not GunHandler or not OriginalGetAim then return end
    GunHandler.getAim = function(origin, range)
        if SilentAim.Enabled then
            local best = getBestSilentTarget()
            if best and best.Part then
                local targetPos = best.Part.Position
                local dir = targetPos - origin
                if dir.Magnitude > 0.001 then
                    return dir.Unit, range
                end
            end
        end
        local ok, a, b = pcall(OriginalGetAim, origin, range)
        if ok and a then return a, b end
        return Camera.CFrame.LookVector, range
    end
end

    if GunHandler and OriginalGetAim then
        GunHandler.getAim = OriginalGetAim
    end
end

    lookAtPos = lookAtPos or (goalPos + Vector3.new(0, 0, -1))
    clearBringPhysicsRep(controlHrp)
    zeroRootMotion(controlHrp)
    pcall(function()

    end)
    local hoodChar = getHoodCharacterModel(LocalPlayer)
    if hoodChar then

        if hoodHrp and hoodHrp ~= controlHrp then

    end
    return (goalPos - controlHrp.Position).Magnitude
end

local function runBringSequence(targetPlayer, token)
    local targetName = targetPlayer.Name
    if status then
        status.Text = "Status: BRING (KNOCK) " .. targetName
    end

    local knockTimeout = tick() + 90
    while token == bringCancelToken and isBringing and tick() < knockTimeout do
        local targetChar = getTargetHoodChar(targetPlayer)
        if targetChar and isCharacterKnocked(targetChar, targetPlayer) then
            break
        end
        currentTarget = targetPlayer
        isOrbiting = true
        isManualTarget = true
        task.wait(0.05)
    end

    if token ~= bringCancelToken or not isBringing then
        finishBring()
        return
    end

    local targetChar = getTargetHoodChar(targetPlayer)
    if not targetChar or not isCharacterKnocked(targetChar, targetPlayer) then
        finishBring()
        return
    end

    isOrbiting = false
    isManualTarget = false
    currentTarget = nil
    unequipAll()
    setRightClick(false)
    if status then
        status.Text = "Status: BRING (GRAB) " .. targetName
    end

    local grabbed = false
    local grabTimeout = tick() + 14
    while token == bringCancelToken and isBringing and tick() < grabTimeout do
        targetChar = getTargetHoodChar(targetPlayer)
        local bodyPart = getKnockedBodyPart(targetChar)
        if not targetChar or not bodyPart or not isCharacterKnocked(targetChar, targetPlayer) then
            break
        end
        local hoverCf = CFrame.new(bodyPart.Position + Vector3.new(0, 3.5, 0), bodyPart.Position + Vector3.new(0, 2, 0))
        setAllLocalHrpsCFrame(hoverCf, true)
        if isLocalGrabbing() then
            grabbed = true
            break
        end
        fireMainEventGrab()
        task.wait(0.35)
    end

    if token ~= bringCancelToken or not isBringing then
        finishBring()
        return
    end
    if not grabbed then
        finishBring()
        return
    end

    if status then
        status.Text = "Status: BRING (TP HOST) " .. targetName
    end
    local hostHrp = getHostHrp()
    if not hostHrp then
        fireMainEventDropGrab()
        finishBring()
        return
    end

    local hostCf = getHostDropCFrame(hostHrp)
    for _ = 1, 10 do
        if token ~= bringCancelToken then
            finishBring()
            return
        end
        hostHrp = getHostHrp()
        if hostHrp then hostCf = getHostDropCFrame(hostHrp) end
        setAllLocalHrpsCFrame(hostCf, true)
        task.wait(0.07)
    end

    if status then
        status.Text = "Status: BRING (DROP) " .. targetName
    end
    task.wait(0.45)
    if token ~= bringCancelToken then
        finishBring()
        return
    end

    unanchorBringControl()
    fireMainEventDropGrab()
    task.wait(0.15)
    fireMainEventDropGrab()
    finishBring()
end

local function playerFromBodyEffectValue(val)
    if not val then return nil end
    if val:IsA("ObjectValue") and val.Value then
        if val.Value:IsA("Player") then return val.Value end
        return Players:GetPlayerFromCharacter(val.Value)
    end
    if val:IsA("StringValue") and val.Value ~= "" then
        return findPlayer(val.Value)
    end
    if val:IsA("IntValue") then
        return Players:GetPlayerByUserId(val.Value)
    end
    return nil
end

local function nameMatchesLocalPlayer(nameStr)
    if not nameStr or nameStr == "" then return false end
    local lower = string.lower(nameStr)
    return lower == string.lower(LocalPlayer.Name) or lower == string.lower(LocalPlayer.DisplayName)
        or string.find(lower, string.lower(LocalPlayer.Name), 1, true) ~= nil
end

local function getAttackerFromBodyEffects(char)
    if not char then return nil end
    local be = char:FindFirstChild("BodyEffects")
    if not be then return nil end

    local priorityKeys = { "Attacker", "LastAttacker", "ShotBy", "LastDamagedBy", "DamagedBy", "HitBy", "Attacking" }
    for i = 1, #priorityKeys do
        local plr = playerFromBodyEffectValue(be:FindFirstChild(priorityKeys[i]))
        if plr and plr ~= LocalPlayer and not isWhitelisted(plr) then return plr end
    end

    local children = be:GetChildren()
    for i = 1, #children do
        local child = children[i]
        local lowerName = string.lower(child.Name)
        if lowerName:find("attack") or lowerName:find("shot") or lowerName:find("damage") or lowerName:find("hit") then
            local plr = playerFromBodyEffectValue(child)
            if plr and plr ~= LocalPlayer and not isWhitelisted(plr) then return plr end
        end
    end
    return nil
end

local function isCombatTool(tool)
    if not tool or not tool:IsA("Tool") then return false end
    local tName = string.lower(tool.Name)
    if tName:find("knife") or tName:find("pitchfork") or tName:find("stopsign") or tName:find("whip") or tName:find("fist") then
        return false
    end
    return tool:FindFirstChild("Handle") ~= nil
end

local function findLikelyRangedAttacker(myHrp)
    if not myHrp then return nil end
    local myPos = myHrp.Position
    local bestPlayer = nil
    local bestScore = -math.huge

    local players = Players:GetPlayers()
    for i = 1, #players do
        local p = players[i]
        if p ~= LocalPlayer and not isWhitelisted(p) and p.Character then
            local pChar = p.Character
            local pHrp = pChar:FindFirstChild("HumanoidRootPart")
            local pHum = pChar:FindFirstChildOfClass("Humanoid")
            if pHrp and pHum and pHum.Health > 0 then
                local dist = (pHrp.Position - myPos).Magnitude
                if dist <= 280 then
                    local toMe = myPos - pHrp.Position
                    local flatToMe = Vector3.new(toMe.X, 0, toMe.Z)
                    if flatToMe.Magnitude > 0.1 then
                        flatToMe = flatToMe.Unit
                        local lookFlat = Vector3.new(pHrp.CFrame.LookVector.X, 0, pHrp.CFrame.LookVector.Z)
                        if lookFlat.Magnitude > 0.1 then
                            lookFlat = lookFlat.Unit
                            local aimDot = lookFlat:Dot(flatToMe)
                            local tool = pChar:FindFirstChildOfClass("Tool")
                            local hasGun = isCombatTool(tool)
                            local attackingMe = false
                            local pBe = pChar:FindFirstChild("BodyEffects")
                            if pBe then
                                local atkVal = pBe:FindFirstChild("Attacking") or pBe:FindFirstChild("AttackingPlayer") or pBe:FindFirstChild("Target")
                                if atkVal and atkVal:IsA("StringValue") and nameMatchesLocalPlayer(atkVal.Value) then
                                    attackingMe = true
                                end
                            end

                            local score = aimDot + (hasGun and 0.75 or 0) + (attackingMe and 2.5 or 0) - (dist / 400)
                            if (attackingMe or (hasGun and aimDot > 0.35) or aimDot > 0.65) and score > bestScore then
                                bestScore = score
                                bestPlayer = p
                            end
                        end
                    end
                end
            end
        end
    end
    return bestPlayer
end

    local myHrp = char:FindFirstChild("HumanoidRootPart")

    isFollowing = false

        stompTargetName = nil

local function bindProtectionCharacter(char)
    if not char then return end

        if not isProtectionActive and not isProtection2Active then return end

        local toucher = getPlayerFromTouchPart(hit)
        if not toucher then return end
        engageProtectionTarget(toucher, isProtection2Active)
    end
    local descendants = char:GetDescendants()
    for i = 1, #descendants do
        local desc = descendants[i]
        if desc:IsA("BasePart") then
            desc.Touched:Connect(onTouched)
        end
    end
    char.DescendantAdded:Connect(function(desc)
        if desc:IsA("BasePart") then
            desc.Touched:Connect(onTouched)
        end
    end)

    local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid", 8)
    if hum then
        lastProtectionHealth = hum.Health
        hum.HealthChanged:Connect(function(newHealth)
            if lastProtectionHealth and newHealth < lastProtectionHealth - 0.25 then
                tryProtectionRetaliationFromDamage()
            end
            lastProtectionHealth = newHealth
        end)
    end

    task.spawn(function()
        local be = char:WaitForChild("BodyEffects", 12)
        if not be then return end
        be.ChildAdded:Connect(function(child)
            if not isProtectionActive and not isProtection2Active then return end
            local lowerName = string.lower(child.Name)
            if lowerName:find("attack") or lowerName:find("shot") or lowerName:find("damage") or lowerName:find("hit") then
                tryProtectionRetaliationFromDamage()
            end
        end)
    end)
end

local function getAltIndex()
    local alts = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if isFormationAlt(p) then alts[#alts + 1] = p end
    end
    table.sort(alts, function(a, b) return a.UserId < b.UserId end)
    for i = 1, #alts do
        if alts[i] == LocalPlayer then return i, #alts end
    end
    return 1, math.max(1, #alts)
end

local function getFormationOffset(mode, altIndex, totalAlts)
    totalAlts = math.max(1, totalAlts)
    altIndex = math.clamp(altIndex, 1, totalAlts)
    local spacing = FORMATION_SPACING

    if mode == "box" then
        local cols = math.max(2, math.ceil(math.sqrt(totalAlts)))
        local index0 = altIndex - 1
        local row = math.floor(index0 / cols)
        local col = index0 % cols
        local slotsBeforeRow = row * cols
        local slotsInRow = math.min(cols, totalAlts - slotsBeforeRow)
        if slotsInRow < 1 then
            slotsInRow = cols
        end
        local rowWidth = (slotsInRow - 1) * spacing
        local x = -rowWidth * 0.5 + col * spacing
        local z = -5.5 - row * spacing
        return Vector3.new(x, 0, z)

    elseif mode == "triangle" then
        local index0 = altIndex - 1
        local row = 0
        local consumed = 0
        while true do
            local rowSlots = row + 1
            if index0 < consumed + rowSlots then
                local posInRow = index0 - consumed
                local rowWidth = (rowSlots - 1) * spacing
                local x = -rowWidth * 0.5 + posInRow * spacing
                local z = -5.5 - row * (spacing * 0.85)
                return Vector3.new(x, 0, z)
            end
            consumed = consumed + rowSlots
            row = row + 1
        end

    elseif mode == "line" then
        local totalWidth = (totalAlts - 1) * spacing
        local x = -totalWidth * 0.5 + (altIndex - 1) * spacing
        return Vector3.new(x, 0, -5)

    elseif mode == "wedge" then
        local row = math.floor((altIndex - 1) / 2)
        local side = ((altIndex - 1) % 2 == 0) and -1 or 1
        return Vector3.new(side * spacing * 0.85, row * 2.1, -5 - row * spacing)

    elseif mode == "surround" or mode == "circle" then
        local radius = math.clamp(4.2 + totalAlts * 0.45, 4.2, 6.5)
        local angle = ((altIndex - 1) / totalAlts) * (2 * math.pi) + (math.pi * 0.5)
        return Vector3.new(math.cos(angle) * radius, 0, math.sin(angle) * radius - 1.2)

    else
        local lane = ((altIndex - 1) % 2 == 0) and -1 or 1
        local laneIndex = math.ceil(altIndex / 2)
        local x = lane * (2 + laneIndex * (spacing * 0.45))
        local z = -4.5 - laneIndex * (spacing * 0.65)
        return Vector3.new(x, 0, z)
    end
end

local function applyFormationFollow(myHrp, leaderHrp, offset)
    local leaderVel = leaderHrp.AssemblyLinearVelocity or Vector3.zero
    local speed = leaderVel.Magnitude
    local predictTime = math.clamp(0.14 + speed / 110, 0.14, 0.38)
    local predictedLeaderPos = leaderHrp.Position + (leaderVel * predictTime)
    local goalCFrame = CFrame.new(predictedLeaderPos) * leaderHrp.CFrame.Rotation * CFrame.new(offset)

    myHrp.AssemblyAngularVelocity = Vector3.zero
    local dist = (goalCFrame.Position - myHrp.Position).Magnitude
    if dist >= FORMATION_SNAP_DISTANCE then
        myHrp.AssemblyLinearVelocity = Vector3.zero
        myHrp.CFrame = goalCFrame
        return
    end

    local alpha = math.clamp(FORMATION_SMOOTH + (dist / 120), 0.22, 0.65)
    myHrp.AssemblyLinearVelocity = Vector3.zero
    myHrp.CFrame = myHrp.CFrame:Lerp(goalCFrame, alpha)
end

local function getGlueCombatGoalCFrame(targetHead, altIndex, totalAlts)
    totalAlts = math.max(1, totalAlts)
    local spread = ((altIndex - 1) / totalAlts) * (2 * math.pi)
    local r = 5.2
    local pos = targetHead.Position + Vector3.new(math.cos(spread) * r, 0, math.sin(spread) * r)
    return CFrame.new(pos, targetHead.Position)
end

local function refreshGlueSpot(myHrp, leaderHrp)
    if not isGlueMovementActive() then glueSpot = nil return end
    local altIndex, totalAlts = getAltIndex()
    if isOrbiting and currentTarget and not isWhitelisted(currentTarget) then
        local tChar = getChar(currentTarget)
        local head = tChar and (tChar:FindFirstChild("Head") or tChar:FindFirstChild("HumanoidRootPart"))
        if head then glueSpot = getGlueCombatGoalCFrame(head, altIndex, totalAlts) end
    elseif leaderHrp and (isFollowing or isGuardActive) then
        if isHiding and not isGuardActive and not isOrbiting and not isStomping then
            glueSpot = nil
            return
        end
        local offset = getFormationOffset(formationMode, altIndex, totalAlts)
        local leaderVel = leaderHrp.AssemblyLinearVelocity or Vector3.zero
        local predictTime = math.clamp(0.1 + leaderVel.Magnitude / 140, 0.1, 0.28)
        local predicted = leaderHrp.Position + leaderVel * predictTime
        glueSpot = CFrame.new(predicted) * leaderHrp.CFrame.Rotation * CFrame.new(offset)
    else
        glueSpot = nil
    end
end

    end
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
    root.CFrame = glueSpot
end

local function executeCommand(speaker, msg)
    if isUnloaded or not speaker or not msg then return end

    local isOwner = (string.lower(speaker.Name) == string.lower(OWNER_NAME) or string.lower(speaker.DisplayName) == string.lower(OWNER_NAME))
    local isAuthorized = isOwner or (grantedUser and speaker == grantedUser)
    if not isAuthorized then return end

    local args = {}
    for word in string.gmatch(msg, "%S+") do table.insert(args, word) end
    local rawCmd = args[1] or ""
    local param = args[2]
    local cleanCmd = string.lower(rawCmd):gsub("^[%./!]+", "")

    if cleanCmd == "eject" or cleanCmd == "unload" then
        if isOwner then ejectScript() end
    elseif cleanCmd == "give" and isOwner then
        local targetUser = findPlayer(param)
        if targetUser then grantedUser = targetUser; status.Text = "Access given to: " .. targetUser.Name
        else status.Text = "Player not found!" end
    elseif cleanCmd == "return" then
        if isOwner or (grantedUser and speaker == grantedUser) then grantedUser = nil; status.Text = "Access returned to owner." end
    elseif cleanCmd == "dance" or cleanCmd == "syncdance" then
        isDanceEnabled = not isDanceEnabled
        if not isDanceEnabled then stopSyncDance() end
        status.Text = "Sync Dance: " .. (isDanceEnabled and "ON" or "OFF")
    elseif cleanCmd == "stopdance" then
        isDanceEnabled = false
        stopSyncDance()
        status.Text = "Sync Dance: OFF"
    elseif cleanCmd == "gkill" or cleanCmd == "gk" then
        toggleGKill()
    elseif cleanCmd == "silent" or cleanCmd == "silentaim" then
        SilentAim.Enabled = not SilentAim.Enabled
        if SilentAim.Enabled then applyGunHandlerHook() else removeGunHandlerHook() end
        status.Text = "Silent Aim: " .. tostring(SilentAim.Enabled)
    elseif cleanCmd == "reset" or cleanCmd == "forcereset" or cleanCmd == "re" then
        fullStopAndReturn()
        if not fireHoodResetCharacter() then
            local char = LocalPlayer.Character
            if char then pcall(function() char:BreakJoints() end) end
        end
    elseif cleanCmd == "loopkill" or cleanCmd == "lk" then
        local t = findPlayer(param)
        if t and not isWhitelisted(t) then
            loopkillTargetName = t.Name
            isLoopkill = true
            isStomping = false
            stompTargetName = nil
            isProtectionRetaliating = false
            isManualTarget = true
            isGuardActive = false
            currentTarget = t
            isOrbiting = true
            isFollowing = false
            isHiding = false
            formationMode = "none"
            status.Text = "Status: LOOPKILLING " .. t.Name
        end
    elseif cleanCmd == "stomp" or cleanCmd == "s" then
        local t = findPlayer(param)
        if t and not isWhitelisted(t) then
            stompTargetName = t.Name
            isStomping = true
            isLoopkill = false
            loopkillTargetName = nil
            isProtectionRetaliating = false
            isManualTarget = true
            isGuardActive = false
            currentTarget = t
            isOrbiting = true
            isFollowing = false
            isHiding = false
            formationMode = "none"
            status.Text = "Status: STOMPING " .. t.Name
        end
    elseif cleanCmd == "knock" or cleanCmd == "k" then
        local t = findPlayer(param)
        if t and not isWhitelisted(t) then
            isLoopkill = false
            loopkillTargetName = nil
            isStomping = false
            stompTargetName = nil
            isProtectionRetaliating = false
            isManualTarget = true
            isGuardActive = false
            currentTarget = t
            isOrbiting = true
            isFollowing = false
            isHiding = false
            formationMode = "none"
            status.Text = "Status: KNOCKING " .. t.Name
        end
    elseif cleanCmd == "protection" or cleanCmd == "prot" then
        local state = parseOnOff(param)
        if state == true then
            isProtectionActive = true
            isProtection2Active = false
            status.Text = "Status: PROTECTION ON"
        elseif state == false then
            isProtectionActive = false
            if isProtectionRetaliating and not isProtection2Active then
                clearProtectionRetaliation("Status: PROTECTION OFF")
            else
                status.Text = "Status: PROTECTION OFF"
            end
        else
            status.Text = "Usage: .protection on/off"
        end
    elseif cleanCmd == "protection2" or cleanCmd == "prot2" then
        local state = parseOnOff(param)
        if state == true then
            isProtection2Active = true
            isProtectionActive = false
            status.Text = "Status: PROTECTION2 ON"
        elseif state == false then
            isProtection2Active = false
            if isProtectionRetaliating then
                clearProtectionRetaliation("Status: PROTECTION2 OFF")
            else
                status.Text = "Status: PROTECTION2 OFF"
            end
        else
            status.Text = "Usage: .protection2 on/off"
        end
    elseif cleanCmd == "b" or cleanCmd == "bring" then
        local t = findPlayer(param)
        if t and not isWhitelisted(t) then
            startBringTarget(t)
        elseif not param or param == "" then
            status.Text = "Usage: .b username"
        else
            status.Text = "Player not found!"
        end
    elseif cleanCmd == "antistomp" or cleanCmd == "as" then
        local state = parseOnOff(param)
        if state == nil and (not param or param == "") then
            isAntiStompActive = not isAntiStompActive
            if isAntiStompActive then resetAntiStompSession() end
            status.Text = "Status: ANTI STOMP " .. (isAntiStompActive and "ON" or "OFF")
        elseif state == true then
            isAntiStompActive = true
            resetAntiStompSession()
            status.Text = "Status: ANTI STOMP ON"
        elseif state == false then
            isAntiStompActive = false
            status.Text = "Status: ANTI STOMP OFF"
        else
            status.Text = "Usage: .antistomp on/off"
        end
    elseif cleanCmd == "stop" or cleanCmd == "unequip" then
        fullStopAndReturn()
    elseif cleanCmd == "knife" then
        attackSubMode = "knife"
        currentWeapon = "Knife"
        equipTool("Knife")
        status.Text = "Weapon: Knife"
    elseif cleanCmd == "revolver" or cleanCmd == "gun" then
        attackSubMode = "combat"
        currentWeapon = "Revolver"
        equipTool("Revolver")
        status.Text = "Weapon: Revolver"
    elseif cleanCmd == "box" then
        formationMode = "box"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Box"
    elseif cleanCmd == "triangle" then
        formationMode = "triangle"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Triangle"
    elseif cleanCmd == "circle" then
        formationMode = "circle"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Circle"
    elseif cleanCmd == "line" then
        formationMode = "line"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Line"
    elseif cleanCmd == "wedge" then
        formationMode = "wedge"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Wedge"
    elseif cleanCmd == "surround" then
        formationMode = "surround"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Surround"
    elseif cleanCmd == "glue" then
        toggleGlueFollow()
    elseif cleanCmd == "hide" then
        stopAndHide()
    elseif cleanCmd == "walk" or cleanCmd == "stand" or cleanCmd == "follow" then
        startStand()
    elseif cleanCmd == "rejoin" or cleanCmd == "rj" then
        TeleportService:Teleport(game.PlaceId, LocalPlayer)
    elseif cleanCmd == "leave" then
        game:Shutdown()
    end
end

local lastAutoFireTick = 0

RunService.RenderStepped:Connect(function()
    if isUnloaded then return end

    local char = getChar(LocalPlayer)
    if not char then return end
    local myHrp = getMoveHrp(LocalPlayer) or char:FindFirstChild("HumanoidRootPart")
    local myHum = getLocalHum() or char:FindFirstChildOfClass("Humanoid")
    if not myHrp or not myHum then return end

    local function applySkyHide()
        myHrp.Anchored = false
        myHrp.AssemblyLinearVelocity = Vector3.zero
        myHrp.AssemblyAngularVelocity = Vector3.zero
        local timeVal = tick() * 15
        local altIndex, totalAlts = getAltIndex()
        local offsetAngle = ((altIndex - 1) / math.max(1, totalAlts)) * (2 * math.pi)
        local rx = math.sin(timeVal + offsetAngle) * 45000
        local rz = math.cos(timeVal * 0.9 + offsetAngle) * 45000
        local ry = 50000 + (math.sin(timeVal * 2 + altIndex) * 3000)
        myHrp.CFrame = CFrame.new(rx, ry, rz)
    end

    local hideCombat = isHiding and (isOrbiting or isStomping or isLoopkill or isBringing or isGuardActive or isManualTarget)

    if (isProtectionActive or isProtection2Active) and not isProtectionRetaliating then
        if lastProtectionHealth and myHum.Health < lastProtectionHealth - 0.25 then
            tryProtectionRetaliationFromDamage()
        end
    end
    lastProtectionHealth = myHum.Health

    -- Safe ADS for guns
    if isHoldingGun() and not isHiding and myHum.Health > 0 then
        setRightClick(true)
    else
        setRightClick(false)
    end

    if isHiding and not hideCombat then
        applySkyHide()
        return
    end

    -- Stomp Mode Check
    if isStomping and stompTargetName and not isBringing then
        local targetPlayer = findPlayer(stompTargetName)
        if targetPlayer and targetPlayer.Character then
            local targetChar = targetPlayer.Character
            if isCharacterKnocked(targetChar) and performKnockStomp(myHrp, targetChar) then
                return
            else
                currentTarget = targetPlayer
                isOrbiting = true
            end
        else
            isStomping = false
            stompTargetName = nil
            if isProtectionRetaliating and isProtection2Active then
                clearProtectionRetaliation("Status: PROTECTION2 ON")
            else
                currentTarget = nil
                isFollowing = true
                status.Text = "Status: FOLLOWING"
            end
        end
    end

    -- Protection (knock only): stop after target is knocked
    if isProtectionActive and isProtectionRetaliating and not isProtection2Active and currentTarget then
        local protChar = currentTarget.Character
        if protChar and isCharacterKnocked(protChar) then
            clearProtectionRetaliation("Status: PROTECTION ON")
            return
        end
    end

    -- Loopkill knock check: stomp when down, resume orbit when they get up
    if isLoopkill and loopkillTargetName and not isBringing then
        local targetPlayer = findPlayer(loopkillTargetName)
        if targetPlayer and targetPlayer.Character then
            currentTarget = targetPlayer
            if isCharacterKnocked(targetPlayer.Character) then
                if performKnockStomp(myHrp, targetPlayer.Character) then
                    return
                end
            else
                isOrbiting = true
            end
        end
    end

    -- Guard / GKILL Mode
    local leader = getLeader()
    local leaderChar = leader and getChar(leader)
    local leaderHrp = leaderChar and leaderChar:FindFirstChild("HumanoidRootPart")

    if isGuardActive and not isManualTarget and not isLoopkill and not isStomping and not isBringing then
        if leaderHrp then
            local nearestEnemy = nil
            local shortestDist = 50.0
            local players = Players:GetPlayers()
            for i = 1, #players do
                local p = players[i]
                local isOwnerCheck = (ownerPlayer and p == ownerPlayer)
                local pChar = getChar(p)
                if p ~= leader and p ~= LocalPlayer and not isOwnerCheck and not isWhitelisted(p) and pChar then
                    local tHrp = pChar:FindFirstChild("HumanoidRootPart")
                    local tHum = pChar:FindFirstChildOfClass("Humanoid")
                    if tHrp and tHum and tHum.Health > 0 and not isCharacterKnocked(pChar, p) then
                        local dist = (tHrp.Position - leaderHrp.Position).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            nearestEnemy = p
                        end
                    end
                end
            end
            if nearestEnemy then
                currentTarget = nearestEnemy
                isOrbiting = true
                isFollowing = false
                formationMode = "none"
            else
                if isOrbiting and currentTarget and not isManualTarget then
                    currentTarget = nil
                    isOrbiting = false
                    isFollowing = true
                end
            end
        end
    end

    -- Combat & Orbiting
    local activeTool = char:FindFirstChildOfClass("Tool")
    local isKnifeMode = (attackSubMode == "knife" or attackSubMode == "pitch" or attackSubMode == "sign" or attackSubMode == "whip")

    if isOrbiting and currentTarget then
        if isWhitelisted(currentTarget) then
            if not isLoopkill and not isManualTarget then
                currentTarget = nil
                isOrbiting = false
                isFollowing = true
            else
                fullStopAndReturn()
            end
            return
        end

        local targetChar = getHoodCharacterModel(currentTarget) or currentTarget.Character
        local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")
        local targetHrp = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
        local targetHead = targetChar and (targetChar:FindFirstChild("Head") or targetHrp)

        if isLoopkill and targetChar and isCharacterKnocked(targetChar, currentTarget) then
            return
        end

        if isBringing and targetChar and isCharacterKnocked(targetChar, currentTarget) then
            return
        end

        if isProtectionActive and isProtectionRetaliating and not isProtection2Active and targetChar and isCharacterKnocked(targetChar) then
            return
        end

        if not targetChar or not targetHum or targetHum.Health <= 0 or not targetHrp or not targetHead then
            if isLoopkill then
                return
            elseif isProtectionRetaliating and (isProtectionActive or isProtection2Active) then
                clearProtectionRetaliation(isProtection2Active and "Status: PROTECTION2 ON" or "Status: PROTECTION ON")
                return
            elseif isBringing then
                return
            elseif not isManualTarget and not isGuardActive and not isStomping then
                currentTarget = nil
                isOrbiting = false
                isFollowing = true
                return
            elseif isGuardActive then
                currentTarget = nil
                isOrbiting = false
                isFollowing = true
                return
            elseif not isStomping then
                fullStopAndReturn()
                return
            end
        end

        if currentWeapon then
            equipTool(currentWeapon)
        end

        Camera.CFrame = CFrame.new(Camera.CFrame.Position, targetHead.Position)

        if activeTool then
            local tName = string.lower(activeTool.Name)
            if tName:find("knife") or tName:find("pitchfork") or tName:find("stopsign") or tName:find("whip") then
                isKnifeMode = true
                pcall(function() activeTool:Activate() end)
            else
                isKnifeMode = false
                local now = tick()
                if now - lastAutoFireTick >= 0.13 then
                    lastAutoFireTick = now
                    pcall(function() activeTool:Activate() end)
                    local mainEvent = ReplicatedStorage:FindFirstChild("MainEvent")
                    if mainEvent and activeTool:FindFirstChild("Handle") then
                        mainEvent:FireServer("UpdateMousePos", targetHead.Position)
                        mainEvent:FireServer(
                            "ShootGun",
                            activeTool.Handle,
                            activeTool.Handle.CFrame.Position,
                            targetHead.Position,
                            targetHead,
                            Vector3.new(0, 0, -1)
                        )
                    end
                end
            end
        end

        local targetPos = targetHead.Position
        local altIndex, totalAlts = getAltIndex()
        totalAlts = math.max(1, totalAlts)
        local spreadAngle = ((altIndex - 1) / totalAlts) * (2 * math.pi)
        local combatRangeRadius = 5.4
        local attackOffset = Vector3.new(math.cos(spreadAngle) * combatRangeRadius, 0, math.sin(spreadAngle) * combatRangeRadius)

        local targetGoalCFrame = CFrame.new(targetPos + attackOffset, targetPos)
        if standPositionMode == "hidden" then
            targetGoalCFrame = CFrame.new(targetPos - Vector3.new(0, 5, 0))
        elseif standPositionMode == "sky" then
            targetGoalCFrame = CFrame.new(targetPos + Vector3.new(0, 20, 0))
        end

        refreshGlueSpot(myHrp, leaderHrp)
        if not (isGlueFollowActive and isGlueMovementActive() and glueSpot) then
            myHrp.AssemblyLinearVelocity = Vector3.zero
            myHrp.CFrame = targetGoalCFrame
        else
            applyGlueToRoot()
        end
    else
        isKnifeMode = false
        if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) and SilentAim.Enabled then
            local best = getBestSilentTarget()
            if best and best.Part then
                Camera.CFrame = CFrame.new(Camera.CFrame.Position, best.Part.Position)
            end
        end
    end

    -- Hitbox Expansion
    local allPlayers = Players:GetPlayers()
    for i = 1, #allPlayers do
        local otherPlayer = allPlayers[i]
        if otherPlayer ~= LocalPlayer and otherPlayer.Character then
            local targetRoot = otherPlayer.Character:FindFirstChild("HumanoidRootPart")
            local targetHum = otherPlayer.Character:FindFirstChildOfClass("Humanoid")
            if targetRoot and targetHum then
                if isKnifeMode and targetHum.Health > 0 and not isWhitelisted(otherPlayer) then
                    targetRoot.Size = Vector3.new(hitboxScaleSize, hitboxScaleSize, hitboxScaleSize)
                    targetRoot.Transparency = 0.5
                    targetRoot.CanCollide = false
                else
                    targetRoot.Size = defaultHitboxSize
                    targetRoot.Transparency = 1
                end
            end
        end
    end

    refreshGlueSpot(myHrp, leaderHrp)
    applyGlueToRoot()

    if not isFollowing or (isHiding and not hideCombat) or isOrbiting then
        if isHiding and not (isOrbiting or isStomping or isLoopkill or isBringing or isGuardActive) then
            applySkyHide()
        end
        return
    end

    if leaderHrp and not (isGlueFollowActive and isGlueMovementActive()) then
        local altIndex, totalAlts = getAltIndex()
        local offset = getFormationOffset(formationMode, altIndex, totalAlts)
        applyFormationFollow(myHrp, leaderHrp, offset)
    end
end)

local function handleChat(speaker, message)
    if isUnloaded or not speaker or not message then return end
    executeCommand(speaker, message)
end

local players = Players:GetPlayers()
for i = 1, #players do
    local p = players[i]
    p.Chatted:Connect(function(msg) handleChat(p, msg) end)
end
Players.PlayerAdded:Connect(function(p)
    p.Chatted:Connect(function(msg) handleChat(p, msg) end)
end)

pcall(function()
    for _, channel in ipairs(TextChatService:GetTextChannels()) do
        channel.MessageReceived:Connect(function(txt)
            if txt.TextSource then
                local sender = Players:GetPlayerByUserId(txt.TextSource.UserId)
                if sender then handleChat(sender, txt.Text) end
            end
        end)
    end
    TextChatService.ChildAdded:Connect(function(child)
        if child:IsA("TextChannel") then
            child.MessageReceived:Connect(function(txt)
                if txt.TextSource then
                    local sender = Players:GetPlayerByUserId(txt.TextSource.UserId)
                    if sender then handleChat(sender, txt.Text) end
                end
            end)
        end
    end)
end)

    if isGlueFollowActive then applyGlueToRoot() end
end)

            resetAntiStompSession()
        end
    end)
end)
