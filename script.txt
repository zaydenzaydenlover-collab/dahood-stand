local function findPlayer(nameQuery)
    if not nameQuery then return nil end
    local query = string.lower(nameQuery)
    local allP = Players:GetPlayers()
    for i = 1, #allP do
        local p = allP[i]
        if string.lower(p.Name) == query or string.lower(p.DisplayName) == query or string.find(string.lower(p.Name), query) or string.find(string.lower(p.DisplayName), query) then
            return p
        end
    end
    return nil
end

local function getChar(player)
    if not player then return nil end
    local folder = workspace:FindFirstChild("Players")
    folder = folder and folder:FindFirstChild("Characters")
    if folder then
        local hood = folder:FindFirstChild(player.Name)
        if hood and hood:IsA("Model") then
            return hood
        end
    end
    return player.Character
end

local function getCharRootPart(char)
    if not char then return nil end
    return char:FindFirstChild("HumanoidRootPart")
        or char:FindFirstChild("UpperTorso")
        or char:FindFirstChild("Torso")
        or char:FindFirstChild("Head")
        or char.PrimaryPart
end

local function getPlayerRootPart(player)
    local c = getChar(player) or player.Character
    return c and getCharRootPart(c), c
end

local function getPartWorldPosition(partOrChar)
    if not partOrChar then return nil end
    if partOrChar:IsA("BasePart") then
        return partOrChar.Position
    end
    local root = getCharRootPart(partOrChar)
    if root then return root.Position end
    for _, d in ipairs(partOrChar:GetDescendants()) do
        if d:IsA("BasePart") then
            return d.Position
        end
    end
    return nil
end

local function collectHoodResetRemotes()
    if hoodResetRemotesCache then return hoodResetRemotesCache end
    local list = {}
    local seen = {}
    local function push(remote)
        if not remote or seen[remote] then return end
        if remote:IsA("RemoteEvent") or remote:IsA("RemoteFunction") then
            seen[remote] = true
            list[#list + 1] = remote
        end
    end
    for i = 1, #HOOD_RESET_REMOTE_NAMES do
        push(ReplicatedStorage:FindFirstChild(HOOD_RESET_REMOTE_NAMES[i]))
    end
    for _, child in ipairs(ReplicatedStorage:GetChildren()) do
        local n = string.lower(child.Name)
        if n:find("main") and (n:find("event") or n:find("remote") or n:find("function")) then
            push(child)
        end
    end
    hoodResetRemotesCache = list
    return list
end

                end
            end)
        end
    end
    pcall(function()

        if mainEvent and mainEvent.FireServer then
            mainEvent:FireServer("ResetCharacter")
            mainEvent:FireServer("Respawn")
        end
    end)
end

local ownerPlayer = nil
local grantedUser = nil

local function getLeader()
    if grantedUser then
        local gChar = getChar(grantedUser)
        if gChar and gChar:FindFirstChild("HumanoidRootPart") then
            return grantedUser
        end
    end
    -- Prefer listed order: first owner who is in-game with a body.
    for i = 1, #OWNER_NAMES do
        local p = findPlayer(OWNER_NAMES[i])
        if p then
            local oChar = getChar(p)
            if oChar and oChar:FindFirstChild("HumanoidRootPart") then
                ownerPlayer = p
                return p
            end
        end
    end
    -- Fallback: any owner in the server (even if knocked / no HRP yet).
    for i = 1, #OWNER_NAMES do
        local p = findPlayer(OWNER_NAMES[i])
        if p then
            ownerPlayer = p
            return p
        end
    end
    return ownerPlayer
end

local function getLeaderHrp()
    local leader = getLeader()
    if not leader then return nil end
    local char = getChar(leader)
    return char and char:FindFirstChild("HumanoidRootPart")
end

local voidHideSavedParts = {}

local function clearVoidHideProtection(char)
    if not char then
        voidHideSavedParts = {}
        return
    end
    for part, saved in pairs(voidHideSavedParts) do
        if part and part.Parent and saved then
            pcall(function()
                part.Size = saved.size
                part.CanCollide = saved.canCollide
                part.Massless = saved.massless
            end)
        end
    end
    voidHideSavedParts = {}
end

local function applyVoidHideProtection(char)
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            if not voidHideSavedParts[part] then
                voidHideSavedParts[part] = {
                    size = part.Size,
                    canCollide = part.CanCollide,
                    massless = part.Massless,
                }
            end
            pcall(function()
                part.Size = C.VOID_HIDE_PART_SIZE
                part.CanCollide = false
                part.Massless = true
            end)
        end
    end
end

local function hoodPerformResetBurst(quickMode, singleShot)
    if resetInProgress then return end
    resetInProgress = true
    glueSpot = nil
    glueDesync = false

    local priorChar = LocalPlayer.Character
    isOrbiting = false
    isHiding = false

    pcall(function()
        clearGluePhysics(getChar(LocalPlayer) and getChar(LocalPlayer):FindFirstChild("HumanoidRootPart"))
    end)
    pcall(function()
        clearVoidHideProtection(getChar(LocalPlayer))
    end)

    isAntiStompActive = false
    disconnectAntiStompListeners()

    local useHard = false
    local attempts = singleShot and 1 or (quickMode and 1 or 2)

    task.spawn(function()
        local ok = pcall(function()
            for i = 1, attempts do
                if isUnloaded then break end
                if LocalPlayer.Character ~= priorChar or not LocalPlayer.Character then
                    break
                end
                hoodResetAttempt(i == 1, useHard)
                if attempts > 1 then
                    task.wait(quickMode and 0.14 or 0.22)
                end
            end
        end)

        local spawned = false
        local conn
        conn = LocalPlayer.CharacterAdded:Connect(function(newChar)
            spawned = true
            if conn then
                conn:Disconnect()
                conn = nil
            end
            task.spawn(function()
                pcall(function() newChar:WaitForChild("Humanoid", 12) end)
                pcall(function() newChar:WaitForChild("HumanoidRootPart", 12) end)
                finishResetSpawn(newChar)
            end)
        end)

        task.wait(singleShot and 4 or 6)
        if conn then
            conn:Disconnect()
            conn = nil
        end

        if not spawned and not isUnloaded then
            if LocalPlayer.Character and LocalPlayer.Character ~= priorChar then
                finishResetSpawn(LocalPlayer.Character)
            else
                resetInProgress = false
                isGlueFollowActive = true
                isAntiStompActive = true
                rearmAntiStompAfterGrace(LocalPlayer.Character)
                if status and status.Parent then
                    status.Text = ok and "Status: RESET (tap Rejoin if stuck)" or "Status: RESET ERR"
                    status.TextColor3 = Color3.fromRGB(255, 180, 100)
                end
            end
        elseif not spawned then
            resetInProgress = false
            isAntiStompActive = true
            rearmAntiStompAfterGrace(LocalPlayer.Character)
        end
    end)
end

local function stopSyncDance()
    if currentDanceTrack then
        pcall(function()
            currentDanceTrack:Stop()
            currentDanceTrack:Destroy()
        end)
        currentDanceTrack = nil
    end
    if danceAnimObj then
        pcall(function() danceAnimObj:Destroy() end)
        danceAnimObj = nil
    end

local function playSyncDance(danceData, targetTimePos)
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end

    if currentDanceTrack then
        pcall(function() currentDanceTrack:Stop(); currentDanceTrack:Destroy() end)
        currentDanceTrack = nil
    end

    local anim = Instance.new("Animation")
    anim.AnimationId = danceData.Id
    danceAnimObj = anim

    local track = hum:LoadAnimation(anim)
    track.Priority = Enum.AnimationPriority.Action
    track.Looped = true
    track:Play()
    if targetTimePos and track.Length > 0 then
        track.TimePosition = targetTimePos % track.Length
    end
    currentDanceTrack = track
end

                local now = workspace:GetServerTimeNow() or os.time()
                local minuteBlock = math.floor(now / 60)
                local targetIndex = (minuteBlock % #SyncDances) + 1

                local leaderTrack = nil
                local leader = getLeader()

                    local lHum = leader.Character:FindFirstChildOfClass("Humanoid")

                                for idx, d in ipairs(SyncDances) do
                                    if d.Id == t.Animation.AnimationId then
                                        targetIndex = idx
                                        leaderTrack = t
                                        break
                                    end
                                end
                            end
                            if leaderTrack then break end
                        end
                    end
                end

                if currentDanceIndex ~= targetIndex or not currentDanceTrack or not currentDanceTrack.IsPlaying then
                    currentDanceIndex = targetIndex
                    local targetTime = leaderTrack and leaderTrack.TimePosition or (now % 5)

                    if math.abs(currentDanceTrack.TimePosition - expectedPos) > 0.35 then
                        currentDanceTrack.TimePosition = expectedPos
                    end
                end
            end
        else
            if currentDanceTrack then
                stopSyncDance()
            end
        end
    end

local function getTargetWorldDistance(targetPlayer)
    if not targetPlayer then return 0 end
    local myPos = getPartWorldPosition(getChar(LocalPlayer))
    local _, tChar = getPlayerRootPart(targetPlayer)
    local tPos = getPartWorldPosition(tChar)
    if not myPos or not tPos then return 0 end
    return (tPos - myPos).Magnitude
end

local function isExtremeWorldPosition(pos)
    if not pos then return false end
    local ax = math.abs(pos.X)
    local ay = math.abs(pos.Y)
    local az = math.abs(pos.Z)
    local maxAxis = math.max(ax, ay, az)
    if maxAxis >= C.VOID_HUNT_ORIGIN_STUDS then return true end
    if pos.Magnitude >= C.VOID_HUNT_ORIGIN_STUDS then return true end
    if ay >= C.VOID_HUNT_FAR_STUDS then return true end
    return false
end

local function isVoidHidingPlayer(player)
    if not player then return false end
    local _, c = getPlayerRootPart(player)
    if not c then return false end
    local pos = getPartWorldPosition(c)
    if not pos then return false end
    if isExtremeWorldPosition(pos) then return true end
    local leaderHrp = getLeaderHrp()
    if leaderHrp and isExtremeWorldPosition(pos - leaderHrp.Position) then
        return true
    end
    local myPos = getPartWorldPosition(getChar(LocalPlayer))
    if myPos then
        local sep = (pos - myPos).Magnitude
        if sep >= C.VOID_HUNT_SEPARATION_STUDS then return true end
        if sep >= C.VOID_HUNT_ORBIT_STUDS and isExtremeWorldPosition(pos) then
            return true
        end
    end
    return false
end

local function shouldVoidHuntTarget(player)
    if not player then return false end
    if isVoidHidingPlayer(player) then return true end
    if getTargetWorldDistance(player) > C.VOID_HUNT_ORBIT_STUDS then return true end
    local pos = getPartWorldPosition(getChar(player) or player.Character)
    if pos and isExtremeWorldPosition(pos) then return true end
    return false
end

local function snapVoidHuntCFrame(hrp, cf, pin)
    if not hrp or not cf then return end
    hrp.Anchored = false
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    if pin and pin.Parent and sethiddenproperty then
        pcall(function() sethiddenproperty(hrp, "PhysicsRepRootPart", pin) end)
        pcall(function()
            hrp.CFrame = pin.CFrame:Inverse() * cf
        end)
    end
    local goalPos = cf.Position
    local scale = math.max(math.abs(goalPos.X), math.abs(goalPos.Y), math.abs(goalPos.Z), goalPos.Magnitude, 1)
    local writes = scale >= C.VOID_HUNT_DEEP_STUDS and C.VOID_HUNT_DEEP_WRITES or C.VOID_HUNT_WRITES
    for _ = 1, writes do
        hrp.CFrame = cf
    end
    pcall(function() hrp:ApplyImpulse(Vector3.new(0, 0.01, 0)) end)
end

local function getBestSilentTarget()
    if not SilentAim.Enabled then return nil end

    if currentTarget and not isWhitelisted(currentTarget) then
        local char = getChar(currentTarget) or currentTarget.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local part = char and (
            shouldUseInfiniteChase(currentTarget) and getVoidHuntAimPart(char) or getAimPart(char)
        )

        if hum and hum.Health > 0 and part then
            local be = char:FindFirstChild("BodyEffects")
            local isKO = be and (
                (be:FindFirstChild("K.O") and be["K.O"].Value == true) or
                (be:FindFirstChild("Dead") and be.Dead.Value == true)
            )
            if not isKO then
                return { Part = part, Character = char }
            end
        end
    end

    local bestTarget = nil
    local shortestDist = SilentAim.FOV
    local mousePos = UserInputService:GetMouseLocation()
    local localHrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not localHrp then return nil end

    local players = Players:GetPlayers()
    for i = 1, #players do
        local player = players[i]
        if player ~= LocalPlayer and not isWhitelisted(player) then
            local char = getChar(player) or player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local part = char and getAimPart(char)

            if hum and hum.Health > 0 and part then
                local worldDist = (part.Position - localHrp.Position).Magnitude
                if worldDist <= getSilentAimMaxDistance() then
                    local be = char:FindFirstChild("BodyEffects")
                    local isKO = be and (
                        (be:FindFirstChild("K.O") and be["K.O"].Value == true) or
                        (be:FindFirstChild("Dead") and be.Dead.Value == true)
                    )

                    if not isKO then
                        local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                        if onScreen then
                            local screen2D = Vector2.new(screenPos.X, screenPos.Y)
                            local dist = (screen2D - mousePos).Magnitude
                            if dist <= shortestDist then
                                shortestDist = dist
                                bestTarget = { Part = part, Character = char }
                            end
                        end
                    end
                end
            end
        end
    end
    return bestTarget
end

    end

local function performInstantKnockReset()
    if shouldSkipAntiStompKnock() then return end
    local now = tick()
    if now - lastAntiStompTick < R.KNOCK_RESET_DEBOUNCE then return end
    lastAntiStompTick = now
    resetGraceUntil = now + R.KNOCK_RESET_GRACE

    glueSpot = nil
    glueDesync = false
    isOrbiting = false
    isHiding = false
    disconnectAntiStompListeners()
    isAntiStompActive = false

    pcall(function()
        clearGluePhysics(getChar(LocalPlayer) and getChar(LocalPlayer):FindFirstChild("HumanoidRootPart"))
    end)
    pcall(function() clearVoidHideProtection(getChar(LocalPlayer)) end)

    pulseHoodResetNow()

    if resetInProgress then
        return
    end

    resetInProgress = true
    local priorChar = LocalPlayer.Character
    local conn
    conn = LocalPlayer.CharacterAdded:Connect(function(newChar)
        if conn then
            conn:Disconnect()
            conn = nil
        end
        task.defer(function()
            pcall(function() newChar:WaitForChild("Humanoid", 8) end)
            if finishResetSpawn then
                finishResetSpawn(newChar)
            end
        end)
    end)

    task.delay(1.25, function()
        if conn then
            conn:Disconnect()
            conn = nil
        end
        if isUnloaded then return end
        if resetInProgress then
            if LocalPlayer.Character and LocalPlayer.Character ~= priorChar then
                if finishResetSpawn then
                    finishResetSpawn(LocalPlayer.Character)
                end
            else
                resetInProgress = false
                isAntiStompActive = true
                if rearmAntiStompAfterGrace then
                    rearmAntiStompAfterGrace(LocalPlayer.Character)
                end
            end
        end
    end)
end

armAntiStomp = function(character)
    if not character or isUnloaded or not isAntiStompActive then return end
    local effects = character:FindFirstChild("BodyEffects")
    if not effects then
        local ok, child = pcall(function() return character:WaitForChild("BodyEffects", 6) end)
        effects = ok and child or nil
    end
    if not effects then return end
    local ko = effects:FindFirstChild("K.O")
    if not ko then
        local ok, child = pcall(function() return effects:WaitForChild("K.O", 6) end)
        ko = ok and child or nil
    end
    if not ko then return end

    local hum = character:FindFirstChildOfClass("Humanoid")
    if hum then
        if antiStompDiedConn then
            antiStompDiedConn:Disconnect()
            antiStompDiedConn = nil
        end
        antiStompDiedConn = hum.Died:Connect(function()
            if shouldSkipAntiStompKnock() then return end
            performInstantKnockReset()
        end)

        if antiStompHealthConn then
            antiStompHealthConn:Disconnect()
            antiStompHealthConn = nil
        end
        antiStompHealthConn = hum.HealthChanged:Connect(function(hp)
            tryAntiStompLowHpReset(character)
        end)
        tryAntiStompLowHpReset(character)
    end

    if antiStompKoConn then
        antiStompKoConn:Disconnect()
        antiStompKoConn = nil
    end

    antiStompKoConn = ko.Changed:Connect(function()
        if shouldSkipAntiStompKnock() then return end
        local dead = effects:FindFirstChild("Dead")
        if not (ko.Parent and ko.Value == true) then return end
        if dead and dead.Value == true then return end
        performInstantKnockReset()
    end)

    if ko.Value == true then
        local dead = effects:FindFirstChild("Dead")
        if not (dead and dead.Value == true) and not shouldSkipAntiStompKnock() then
            performInstantKnockReset()
        end
    end
end

local function setAntiStomp(active)
    isAntiStompActive = active
    if not active then
        if antiStompKoConn then
            antiStompKoConn:Disconnect()
            antiStompKoConn = nil
        end
        if antiStompDiedConn then
            antiStompDiedConn:Disconnect()
            antiStompDiedConn = nil
        end
        if antiStompHealthConn then
            antiStompHealthConn:Disconnect()
            antiStompHealthConn = nil
        end
    end
    if active then
        local c = getChar(LocalPlayer)
        if c and armAntiStomp then
            task.spawn(function()
                armAntiStomp(c)
            end)
        end
    end
    status.Text = "Anti stomp: " .. (active and "ON" or "OFF") .. " (reset HP<=" .. R.ANTI_STOMP_HP_RESET_AT .. ")"
end

local function getToolAmmoValues(tool)
    if not tool then return nil, nil end
    local ammoVal = tool:FindFirstChild("Ammo", true)
    local maxAmmoVal = tool:FindFirstChild("MaxAmmo", true)
    if ammoVal and not (ammoVal:IsA("IntValue") or ammoVal:IsA("NumberValue")) then
        ammoVal = nil
    end
    if maxAmmoVal and not (maxAmmoVal:IsA("IntValue") or maxAmmoVal:IsA("NumberValue")) then
        maxAmmoVal = nil
    end
    return ammoVal, maxAmmoVal
end

local function triggerReload(tool, ammoVal, maxAmmoVal)
    local currentTime = tick()
    if currentTime - lastReloadTick < 0.35 then return end
    lastReloadTick = currentTime
    task.spawn(function()
        local mainEvent = ReplicatedStorage:FindFirstChild("MainEvent")
        if mainEvent and tool then
            pcall(function() mainEvent:FireServer("Reload", tool) end)
            pcall(function() mainEvent:FireServer("Reload") end)
        elseif mainEvent then
            pcall(function() mainEvent:FireServer("Reload") end)
        end
        pcall(function()
            VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.R, false, game)
            task.wait(0.06)
            VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.R, false, game)
        end)
        if ammoVal and ammoVal.Value <= 0 then
            pcall(function()
                local refill = (maxAmmoVal and maxAmmoVal.Value > 0) and maxAmmoVal.Value or 6
                ammoVal.Value = refill
            end)
        end
    end)
end

local function checkAutoReloadOnTool(tool)
    if not tool then return end
    local ammoVal, maxAmmoVal = getToolAmmoValues(tool)
    if ammoVal and ammoVal.Value <= 0 then
        triggerReload(tool, ammoVal, maxAmmoVal)
    end
end

local function setupToolReloadListener(tool)
    if not tool or reloadHookedTools[tool] then return end
    reloadHookedTools[tool] = true
    local ammoVal, maxAmmoVal = getToolAmmoValues(tool)
    if not ammoVal then return end
    ammoVal.Changed:Connect(function(newVal)
        if newVal <= 0 then triggerReload(tool, ammoVal, maxAmmoVal) end
    end)
    if ammoVal.Value <= 0 then
        triggerReload(tool, ammoVal, maxAmmoVal)
    end
end

local function getAltIndex()
    local host = getLeader()
    local alts = {}
    local seen = {}
    local function push(p)
        if not p or seen[p.UserId] then return end
        if host and p == host then return end
        seen[p.UserId] = true
        alts[#alts + 1] = p
    end
    push(LocalPlayer)
    for _, p in ipairs(Players:GetPlayers()) do
        if isFormationAlt(p) or isArmyMember(p) or playerHasStandoLoaded(p) then
            push(p)
        end
    end
    table.sort(alts, function(a, b) return a.UserId < b.UserId end)
    local n = #alts
    if n < 1 then
        return (math.abs(LocalPlayer.UserId) % 8) + 1, 8
    end
    for i = 1, n do
        if alts[i] == LocalPlayer then
            return i, n
        end
    end
    return (math.abs(LocalPlayer.UserId) % n) + 1, n
end

local function getNearestThreatRangeStuds(myHrp)
    if not myHrp then return math.huge end
    local nearest = math.huge
    local allP = Players:GetPlayers()
    for i = 1, #allP do
        local p = allP[i]
        if p ~= LocalPlayer and not isWhitelisted(p) then
            local c = getChar(p)
            local hrp = c and c:FindFirstChild("HumanoidRootPart")
            if hrp then
                local d = (hrp.Position - myHrp.Position).Magnitude
                if d < nearest then
                    nearest = d
                end
            end
        end
    end
    return nearest
end

local function applyVoidHideCFrame(myHrp)
    if not myHrp then return end
    myHrp.Anchored = false
    myHrp.AssemblyLinearVelocity = Vector3.zero
    myHrp.AssemblyAngularVelocity = Vector3.zero
    local altIndex, totalAlts = getAltIndex()
    local timeVal = tick() * C.VOID_HIDE_SPIN
    local offsetAngle = ((altIndex - 1) / math.max(1, totalAlts)) * (2 * math.pi)
    local rx = math.sin(timeVal + offsetAngle) * C.VOID_HIDE_RX
    local rz = math.cos(timeVal * 0.9 + offsetAngle) * C.VOID_HIDE_RZ
    local ry = C.VOID_HIDE_RY_BASE + (math.sin(timeVal * 2 + altIndex) * C.VOID_HIDE_RY_AMP)
    rx = rx + math.random(-C.VOID_HIDE_JITTER, C.VOID_HIDE_JITTER)
    rz = rz + math.random(-C.VOID_HIDE_JITTER, C.VOID_HIDE_JITTER)
    ry = ry + math.random(-C.VOID_HIDE_JITTER, C.VOID_HIDE_JITTER)
    local threatRange = getNearestThreatRangeStuds(myHrp)
    if threatRange < C.VOID_KNIFE_PANIC_RANGE then
        local push = C.VOID_PANIC_PUSH + (C.VOID_KNIFE_PANIC_RANGE - threatRange) * 150000
        rx = rx + push * (0.6 + math.random())
        rz = rz + push * (0.6 + math.random())
        ry = ry + push * (0.35 + math.random() * 0.5)
    end
    local cf = CFrame.new(rx, ry, rz)
    for _ = 1, C.VOID_HIDE_WRITES do
        myHrp.CFrame = cf
    end
end

local function getHostFlatBasis(leaderHrp)
    local center = leaderHrp.Position
    local look = leaderHrp.CFrame.LookVector
    local flatLook = Vector3.new(look.X, 0, look.Z)
    if flatLook.Magnitude < 0.05 then
        flatLook = Vector3.new(0, 0, -1)
    else
        flatLook = flatLook.Unit
    end
    local flatRight = Vector3.new(-flatLook.Z, 0, flatLook.X)
    if flatRight.Magnitude < 0.05 then
        flatRight = Vector3.new(1, 0, 0)
    else
        flatRight = flatRight.Unit
    end
    return center, flatLook, flatRight
end

local function formationLocalToWorld(leaderHrp, localX, localZ)
    local center, flatLook, flatRight = getHostFlatBasis(leaderHrp)
    return center + flatRight * localX + flatLook * localZ
end

local function formationCFrameFaceForward(leaderHrp, goalPos)
    local _, flatLook = getHostFlatBasis(leaderHrp)
    return CFrame.new(goalPos, goalPos + flatLook)
end

local function formationCFrameFaceHost(leaderHrp, goalPos)
    local center = leaderHrp.Position
    local toHost = Vector3.new(center.X - goalPos.X, 0, center.Z - goalPos.Z)
    if toHost.Magnitude < 0.05 then
        local look = leaderHrp.CFrame.LookVector

    end

local function getPerimeterPoint2D(altIndex, totalAlts, edgeLengths)
    local perimeter = 0
    for i = 1, #edgeLengths do
        perimeter = perimeter + edgeLengths[i]
    end
    local dist = ((altIndex - 1) + 0.5) / totalAlts * perimeter
    for i = 1, #edgeLengths do
        local len = edgeLengths[i]
        if dist <= len then
            return i, dist / len
        end
        dist = dist - len
    end
    return #edgeLengths, 1
end

local function lerp2(a, b, t)
    return a + (b - a) * t
end

local function getBoxLocalPosition(altIndex, totalAlts, halfW, halfD)
    local w, d = halfW * 2, halfD * 2
    local edgeIdx, t = getPerimeterPoint2D(altIndex, totalAlts, { w, d, w, d })
    if edgeIdx == 1 then
        return lerp2(-halfW, halfW, t), -halfD
    elseif edgeIdx == 2 then
        return halfW, lerp2(-halfD, halfD, t)
    elseif edgeIdx == 3 then
        return lerp2(halfW, -halfW, t), halfD
    end
    return -halfW, lerp2(halfD, -halfD, t)
end

local function lerpVec3(a, b, t)
    return a + (b - a) * t
end

local function getTriangleVertices(leaderHrp, side)
    side = math.max(5, side)
    local center, flatLook, flatRight = getHostFlatBasis(leaderHrp)
    local h = side * math.sqrt(3) / 2
    -- Apex in front of host; base behind (host at triangle center)
    local vApex = center + flatLook * (2 * h / 3)
    local vLeft = center - flatLook * (h / 3) - flatRight * (side / 2)
    local vRight = center - flatLook * (h / 3) + flatRight * (side / 2)
    return vApex, vLeft, vRight, side
end

local function getTriangleWorldPosition(leaderHrp, altIndex, totalAlts, side)
    local vApex, vLeft, vRight, s = getTriangleVertices(leaderHrp, side)
    if totalAlts == 1 then
        return vApex
    end
    if totalAlts == 2 then
        if altIndex == 1 then return vLeft end
        return vRight
    end
    if totalAlts == 3 then
        if altIndex == 1 then return vApex end
        if altIndex == 2 then return vLeft end
        return vRight
    end
    local edgeIdx, t = getPerimeterPoint2D(altIndex, totalAlts, { s, s, s })
    if edgeIdx == 1 then
        return lerpVec3(vApex, vRight, t)
    elseif edgeIdx == 2 then
        return lerpVec3(vRight, vLeft, t)
    end
    return lerpVec3(vLeft, vApex, t)
end

-- Host = apex; alts fill rows behind in a V (pair left/right, odd alt centers last row).
local function getWedgeWorldPosition(leaderHrp, altIndex, totalAlts)
    local center, flatLook, flatRight = getHostFlatBasis(leaderHrp)
    local step = C.FORMATION_WEDGE_STEP
    local spread = C.FORMATION_WEDGE_SPREAD
    totalAlts = math.max(1, totalAlts)
    altIndex = math.clamp(altIndex, 1, totalAlts)

    local depth = 1
    local slot = 1
    while slot <= totalAlts do
        local back = depth * step
        if slot + 1 <= totalAlts then
            local wing = back * spread
            if altIndex == slot then
                return center - flatLook * back - flatRight * wing
            end
            if altIndex == slot + 1 then
                return center - flatLook * back + flatRight * wing
            end
            slot = slot + 2
            depth = depth + 1
        else
            if altIndex == slot then
                return center - flatLook * back
            end
            slot = slot + 1
            depth = depth + 1
        end
    end

    return center - flatLook * step
end

local function getSurroundGoalCFrame(leaderHrp, altIndex, totalAlts, mode)
    totalAlts = math.max(1, totalAlts)
    altIndex = math.clamp(altIndex, 1, totalAlts)
    mode = mode or formationMode

    if mode == "box" then
        local lx, lz = getBoxLocalPosition(altIndex, totalAlts, C.FORMATION_BOX_HALF_W, C.FORMATION_BOX_HALF_D)
        return formationCFrameFaceHost(leaderHrp, formationLocalToWorld(leaderHrp, lx, lz))
    end

    if mode == "triangle" then
        local goalPos = getTriangleWorldPosition(leaderHrp, altIndex, totalAlts, C.FORMATION_TRI_SIDE)
        return formationCFrameFaceHost(leaderHrp, goalPos)
    end

    if mode == "wedge" or mode == "v" then
        local goalPos = getWedgeWorldPosition(leaderHrp, altIndex, totalAlts)
        return formationCFrameFaceForward(leaderHrp, goalPos)
    end

    local radius = math.clamp(C.FORMATION_RING_RADIUS + (totalAlts - 1) * 0.15, 4.5, 7)
    if mode ~= "circle" and mode ~= "shield" and mode ~= "none" and mode ~= "sit" then
        mode = "circle"
    end
    if mode == "sit" then
        mode = "circle"
    end
    if mode == "shield" then
        radius = C.FORMATION_SHIELD_RADIUS
    end

    local center = leaderHrp.Position
    local look = leaderHrp.CFrame.LookVector
    local flatLook = Vector3.new(look.X, 0, look.Z)
    if flatLook.Magnitude < 0.05 then
        flatLook = Vector3.new(0, 0, -1)
    else
        flatLook = flatLook.Unit
    end
    local baseYaw = math.atan2(flatLook.X, flatLook.Z)
    local angle = baseYaw + ((altIndex - 1) / totalAlts) * (2 * math.pi)

    local goalPos = center + Vector3.new(math.sin(angle) * radius, 0, math.cos(angle) * radius)
    if mode == "shield" then
        local outward = Vector3.new(goalPos.X - center.X, 0, goalPos.Z - center.Z)
        if outward.Magnitude < 0.05 then
            outward = flatLook
        else
            outward = outward.Unit
        end
        return CFrame.new(goalPos, goalPos + outward)
    end
    return formationCFrameFaceHost(leaderHrp, goalPos)
end

local function getFormationOffset(mode, altIndex, totalAlts)
    totalAlts = math.max(1, totalAlts)
    altIndex = math.clamp(altIndex, 1, totalAlts)
    if mode == "line" then
        local spacing = C.FORMATION_SPACING
        local totalWidth = (totalAlts - 1) * spacing
        return Vector3.new(-totalWidth * 0.5 + (altIndex - 1) * spacing, 0, -3)
    end
    return Vector3.new(0, 0, -3)
end

local function applyFormationFollow_inner(myHrp, leaderHrp, mode, altIndex, totalAlts)
    local goalCFrame
    if mode == "line" then
        local offset = getFormationOffset(mode, altIndex, totalAlts)
        goalCFrame = CFrame.new(leaderHrp.Position) * leaderHrp.CFrame.Rotation * CFrame.new(offset)
    else
        goalCFrame = getSurroundGoalCFrame(leaderHrp, altIndex, totalAlts, mode)
    end

    myHrp.AssemblyAngularVelocity = Vector3.zero
    local dist = (goalCFrame.Position - myHrp.Position).Magnitude
    if dist >= C.FORMATION_SNAP_DISTANCE then
        myHrp.AssemblyLinearVelocity = Vector3.zero
        myHrp.CFrame = goalCFrame
        return
    end

    local alpha = math.clamp(C.FORMATION_SMOOTH + (dist / 50), 0.55, 0.92)
    myHrp.AssemblyLinearVelocity = Vector3.zero
    myHrp.CFrame = myHrp.CFrame:Lerp(goalCFrame, alpha)
end

local function getGlueGoalCFrame_inner(leaderHrp, mode, altIndex, totalAlts)
    if mode == "line" then
        local offset = getFormationOffset(mode, altIndex, totalAlts)
        return CFrame.new(leaderHrp.Position) * leaderHrp.CFrame.Rotation * CFrame.new(offset)
    end
    return getSurroundGoalCFrame(leaderHrp, altIndex, totalAlts, mode)
end
applyFormationFollow = applyFormationFollow_inner
getGlueGoalCFrame = getGlueGoalCFrame_inner
end)()

local function buildCombatOrbitCFrame(targetPosOrPart, altIndex, totalAlts)
    if not targetPosOrPart then return nil end
    totalAlts = math.max(1, totalAlts)
    altIndex = math.clamp(altIndex or 1, 1, totalAlts)
    local orbitPart = (typeof(targetPosOrPart) == "Instance") and targetPosOrPart or nil
    local targetPos = orbitPart and orbitPart.Position or targetPosOrPart
    if not targetPos then return nil end
    local baseCF = orbitPart and orbitPart.CFrame or CFrame.new(targetPos)
    local orbitRadius = (isStacking and C.STACK_ORBIT_RADIUS) or C.COMBAT_ORBIT_RADIUS
    local scale = math.max(math.abs(targetPos.X), math.abs(targetPos.Y), math.abs(targetPos.Z), targetPos.Magnitude, 1)
    local deepVoid = scale >= C.VOID_HUNT_DEEP_STUDS
    local extremeVoid = scale >= C.VOID_HUNT_ORIGIN_STUDS
    local spreadAngle = ((altIndex - 1) / totalAlts) * (2 * math.pi)

    -- Deep / extreme void: attach in target local space (PhysicsRepRootPart carries the offset).
    -- Absolute world offsets vanish to float noise past ~tens of millions of studs.
    if deepVoid or extremeVoid then
        if standPositionMode == "hidden" then
            return baseCF * CFrame.new(0, deepVoid and -1.6 or -3.2, 0)
        end
        local attachR = deepVoid and C.VOID_HUNT_ATTACH_RADIUS or C.VOID_ORBIT_MIN_RADIUS
        if isStacking then
            attachR = math.min(attachR, C.STACK_ORBIT_RADIUS)
        end
        local yLift = (altIndex - 1) * (deepVoid and 0.45 or C.VOID_ORBIT_Y_STEP)
        if standPositionMode == "above" or standPositionMode == "sky" then
            yLift = yLift + (deepVoid and 2.4 or C.COMBAT_ORBIT_HEIGHT)
        else
            yLift = yLift + (deepVoid and 1.15 or 2.5)
        end
        if not deepVoid then
            local floatSafe = math.max(C.COMBAT_ORBIT_RADIUS, scale * (2 ^ -20) * 4)
            attachR = math.max(attachR, math.min(floatSafe, 80))
            attachR = attachR + (altIndex - 1) * math.max(1.8, attachR * 0.08)
        else
            attachR = attachR + (altIndex - 1) * 0.55
        end
        return baseCF * CFrame.new(
            math.cos(spreadAngle) * attachR,
            yLift,
            math.sin(spreadAngle) * attachR
        )
    end

    if standPositionMode == "hidden" then
        local hideLift = math.max(5, orbitRadius * 0.02)
        return CFrame.new(targetPos - Vector3.new(0, hideLift, 0), targetPos)
    end
    local yLift = 0
    if standPositionMode == "above" or standPositionMode == "sky" then
        yLift = C.COMBAT_ORBIT_HEIGHT
    end
    yLift = yLift + (altIndex - 1) * 0.35
    local attackOffset = Vector3.new(
        math.cos(spreadAngle) * orbitRadius,
        yLift,
        math.sin(spreadAngle) * orbitRadius
    )
    local goalPos = targetPos + attackOffset
    return CFrame.new(goalPos, targetPos)
end

local function getGlueCombatGoalCFrame(orbitPart, altIndex, totalAlts)
    if not orbitPart then return nil end
    return buildCombatOrbitCFrame(orbitPart, altIndex, totalAlts)
end

local function getGlueStompGoalCFrame(torso, altIndex, totalAlts)
    totalAlts = math.max(1, totalAlts)
    altIndex = math.clamp(altIndex, 1, totalAlts)
    local spreadAngle = ((altIndex - 1) / totalAlts) * (2 * math.pi)
    local ring = 1.15
    local tPos = torso.Position
    local scale = math.max(math.abs(tPos.X), math.abs(tPos.Y), math.abs(tPos.Z), tPos.Magnitude, 1)
    if scale >= C.VOID_HUNT_ORIGIN_STUDS then
        local y = scale >= C.VOID_HUNT_DEEP_STUDS and 2.4 or 3.2
        local r = scale >= C.VOID_HUNT_DEEP_STUDS and 0.9 or ring
        return torso.CFrame * CFrame.new(math.cos(spreadAngle) * r, y, math.sin(spreadAngle) * r)
    end
    local base = tPos + Vector3.new(0, 3.2, 0)
    local pos = base + Vector3.new(math.cos(spreadAngle) * ring, 0, math.sin(spreadAngle) * ring)
    return CFrame.new(pos, base)
end

local function runGlueHeartbeat()
    if not isGlueMovementActive() then return end
    local myChar = getChar(LocalPlayer)
    local root = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local pin, th
    local altIndex, totalAlts = getAltIndex()
    local gluedStomp = false

    local bodyHoverName = (isStomping and stompTargetName) or (isBringing and bringTargetName and (bringPhase == "hover" or bringPhase == "return" or bringPhase == "drop_wait") and bringTargetName)
    if bodyHoverName then
        local bodyPlayer = findPlayer(bodyHoverName)
        if bodyPlayer and not isWhitelisted(bodyPlayer) then
            local theirChar = getChar(bodyPlayer)
            if theirChar then
                local targetBE = theirChar:FindFirstChild("BodyEffects")
                local targetHum = theirChar:FindFirstChildOfClass("Humanoid")
                local knocked = (targetBE and targetBE:FindFirstChild("K.O") and targetBE["K.O"].Value == true)
                    or (targetHum and targetHum.Health <= 0)
                if knocked and isStomping then
                    local torso = theirChar:FindFirstChild("UpperTorso") or theirChar:FindFirstChild("Torso") or theirChar:FindFirstChild("HumanoidRootPart")
                    if torso then
                        pin = torso.AssemblyRootPart or torso
                        glueSpot = getGlueStompGoalCFrame(torso, altIndex, totalAlts)
                        gluedStomp = true
                    end
                elseif knocked and isBringing and bringPhase == "hover" then
                    local torso = theirChar:FindFirstChild("UpperTorso") or theirChar:FindFirstChild("Torso") or theirChar:FindFirstChild("HumanoidRootPart")
                    if torso then
                        pin = torso.AssemblyRootPart or torso
                        glueSpot = getGlueStompGoalCFrame(torso, altIndex, totalAlts)
                        gluedStomp = true
                    end
                elseif isBringing and bringPhase == "drop_wait" then
                    local hostGoal = getHostTeleportCFrame()
                    if hostGoal then
                        pin = getLeaderHrp()
                        pin = pin and (pin.AssemblyRootPart or pin)
                        glueSpot = hostGoal
                        gluedStomp = true
                    end
                end
            end
        end
    end

    if not gluedStomp then
        if isOrbiting and currentTarget and not isWhitelisted(currentTarget) then
            local theirChar = getChar(currentTarget) or currentTarget.Character
            th = theirChar and theirChar:FindFirstChild("HumanoidRootPart")
            local orbitPart = getOrbitPart(theirChar)
            if th and orbitPart then
                local torso = theirChar:FindFirstChild("UpperTorso") or theirChar:FindFirstChild("Torso")
                pin = (torso and torso.AssemblyRootPart) or th
                glueSpot = getGlueCombatGoalCFrame(orbitPart, altIndex, totalAlts)
            end
        elseif isFollowing and not isStomping and not isBringing then
            local host = getLeader()
            if not host or host == LocalPlayer then return end
            local theirChar = getChar(host)
            th = theirChar and theirChar:FindFirstChild("HumanoidRootPart")
            if not th then return end
            local torso = theirChar:FindFirstChild("UpperTorso") or theirChar:FindFirstChild("Torso")
            pin = (torso and torso.AssemblyRootPart) or th
            glueSpot = getGlueGoalCFrame(th, formationMode, altIndex, totalAlts)
        elseif isStomping or (isBringing and bringPhase == "knock") then
            local chaseName = isStomping and stompTargetName or bringTargetName
            local chasePlayer = chaseName and findPlayer(chaseName)
            if chasePlayer and not isWhitelisted(chasePlayer) then
                local theirChar = getChar(chasePlayer)
                th = theirChar and theirChar:FindFirstChild("HumanoidRootPart")
                local orbitPart = getOrbitPart(theirChar)
                if th and orbitPart then
                    local torso = theirChar:FindFirstChild("UpperTorso") or theirChar:FindFirstChild("Torso")
                    pin = (torso and torso.AssemblyRootPart) or th
                    glueSpot = getGlueCombatGoalCFrame(orbitPart, altIndex, totalAlts)
                end
            end
        else
            return
        end
    end
    if not pin then return end

    glueReal, glueVelocity, glueSpin = root.CFrame, root.AssemblyLinearVelocity, root.AssemblyAngularVelocity
    if sethiddenproperty then
        pcall(function() sethiddenproperty(root, "PhysicsRepRootPart", pin) end)
    end
    root.CFrame = pin.CFrame:Inverse() * glueSpot
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
    pcall(function() root:ApplyImpulse(Vector3.new(0, 0.01, 0)) end)
end

local function sendAltChat(text)
    text = tostring(text or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if text == "" then return false end
    if #text > 200 then
        text = string.sub(text, 1, 200)
    end
    local sent = false
    pcall(function()
        local channels = TextChatService:FindFirstChild("TextChannels")
        local general = channels and channels:FindFirstChild("RBXGeneral")
        if general and general.SendAsync then
            general:SendAsync(text)
            sent = true
        end
    end)
    if not sent then
        pcall(function()
            local chatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
            if chatEvents then
                chatEvents:FireServer("SayMessageRequest", text, "All")
                sent = true
            end
        end)
    end
    if not sent then
        pcall(function()
            LocalPlayer:Chat(text)
        end)
    end
    return true
end

local function getCommandTargetQuery(args, startAt)
    startAt = startAt or 2
    if #args < startAt then return nil end
    return table.concat(args, " ", startAt)
end

local function normalizeCmdToken(token)
    if not token or token == "" then return "" end
    return string.lower(token):gsub("^[%./!\\/#:]+", ""):gsub("[%,;:]+$", "")
end

local function isStopCmdToken(token)
    local clean = normalizeCmdToken(token)
    if clean == "" or clean == "stopdance" then return false end
    if clean == "stop" or clean == "stopp" or clean == "stp" or clean == "halt"
        or clean == "unequip" or clean == "fullstop" or clean == "cancel" or clean == "abort"
        or clean == "uns" or clean == "unstomp" or clean == "unst" or clean == "end" then
        return true
    end
    return clean:match("^stop+p?$") ~= nil
end

local function messageRequestsStop(message)
    if not message or message == "" then return false end
    local trimmed = string.lower(message):gsub("^[%./!\\/#%s]+", ""):gsub("%s+$", "")
    if trimmed == "" or trimmed == "stopdance" then return false end
    if isStopCmdToken(trimmed) then return true end
    local first = trimmed:match("^(%S+)")
    return first and isStopCmdToken(first)
end

local function resolveChatSender(textSource)
    if textSource and textSource.UserId then
        local byId = Players:GetPlayerByUserId(textSource.UserId)
        if byId then return byId end
    end
    if textSource and textSource.Name and textSource.Name ~= "" then
        return findPlayer(textSource.Name)
    end
    return nil
end

local function executeCommand(speaker, msg)
    if isUnloaded or not msg or not speaker then return end

    local args = {}
    for word in string.gmatch(msg, "%S+") do table.insert(args, word) end
    local rawCmd = args[1] or ""
    local cleanCmdEarly = normalizeCmdToken(rawCmd)
    local param = args[2]
    local cleanCmd = normalizeCmdToken(rawCmd)
    local targetQuery, resetTarget, targetUser, cmdTarget, cmdOk, cmdMsg
    local pLower, sayText, glueTarget, asTarget, host, altIndex

    if cleanCmdEarly == "reset" or cleanCmdEarly == "forcereset" or cleanCmdEarly == "fr" then
        if speaker == LocalPlayer then
            targetQuery = getCommandTargetQuery(args, 2)
            if not targetQuery or targetQuery == "" then
                performForceReset()
                return
            end
            resetTarget = findPlayer(targetQuery)
            if resetTarget == LocalPlayer then
                performForceReset()
                return
            end
        end
    end

    local isOwner = isOwnerPlayer(speaker)
    local isAuthorized = isOwner or (grantedUser and speaker == grantedUser)
    if not isAuthorized then return end

    if messageRequestsStop(msg) then
        fullStopAndReturn()
        return
    end

    if cleanCmd == "eject" or cleanCmd == "unload" then
        if isOwner then ejectScript() end
    elseif cleanCmd == "give" and isOwner then
        targetUser = findPlayer(param)
        if targetUser then grantedUser = targetUser; status.Text = "Access given to: " .. targetUser.Name
        else status.Text = "Player not found!" end
    elseif cleanCmd == "return" then
        if isOwner or (grantedUser and speaker == grantedUser) then grantedUser = nil; status.Text = "Access returned to owner." end
    elseif (cleanCmd == "wl" or cleanCmd == "whitelist") and isOwner then
        cmdOk, cmdMsg = addRuntimeWhitelist(findPlayer(param))
        status.Text = cmdOk and cmdMsg or (cmdMsg or "WL: player not found!")
    elseif (cleanCmd == "unwl" or cleanCmd == "unwhitelist") and isOwner then
        cmdOk, cmdMsg = removeRuntimeWhitelist(findPlayer(param))
        status.Text = cmdOk and cmdMsg or (cmdMsg or "WL: player not found!")
    elseif (cleanCmd == "army" or cleanCmd == "alt") and isOwner then
        cmdOk, cmdMsg = addRuntimeArmy(findPlayer(param))
        status.Text = cmdOk and cmdMsg or (cmdMsg or "Army: player not found!")
    elseif (cleanCmd == "unarmy" or cleanCmd == "unalt") and isOwner then
        cmdOk, cmdMsg = removeRuntimeArmy(findPlayer(param))
        status.Text = cmdOk and cmdMsg or (cmdMsg or "Army: player not found!")
    elseif cleanCmd == "dance" or cleanCmd == "syncdance" then
        isDanceEnabled = not isDanceEnabled
        if not isDanceEnabled then stopSyncDance() end
        status.Text = "Sync Dance: " .. (isDanceEnabled and "ON" or "OFF")
    elseif cleanCmd == "stopdance" then
        isDanceEnabled = false
        stopSyncDance()
        status.Text = "Sync Dance: OFF"
    elseif isStopCmdToken(cleanCmd) then
        fullStopAndReturn()
    elseif cleanCmd == "gkill" or cleanCmd == "gk" then
        toggleGKill()
    elseif cleanCmd == "silent" or cleanCmd == "silentaim" then
        SilentAim.Enabled = not SilentAim.Enabled
        if SilentAim.Enabled then applyGunHandlerHook() else removeGunHandlerHook() end

    elseif cleanCmd == "reset" or cleanCmd == "forcereset" or cleanCmd == "fr" then

    elseif cleanCmd == "loopkill" or cleanCmd == "lk" then

        if cmdTarget and not isWhitelisted(cmdTarget) then
            isStacking = false
            stackTargetName = nil

            isLoopkill = true
            isStomping = false
            stompTargetName = nil
            isBringing = false

            bringPhase = nil
            isManualTarget = true
            isGuardActive = false

            formationMode = "none"
            beginCombatChase(cmdTarget)
            status.Text = "Status: CHASE LK -> " .. cmdTarget.Name
        end
    elseif cleanCmd == "stack" or cleanCmd == "stk" or cleanCmd == "unstack" then
        targetQuery = getCommandTargetQuery(args, 2)
        if cleanCmd == "unstack" or not targetQuery or targetQuery == "" then
            isStacking = false
            stackTargetName = nil
            if isOrbiting and isManualTarget and not isLoopkill and not isStomping and not isBringing then
                fullStopAndReturn()
            else
                status.Text = "Stack: OFF"
            end
        else
            cmdTarget = findPlayer(targetQuery)
            if cmdTarget and not isWhitelisted(cmdTarget) then
                stackTargetName = cmdTarget.Name
                isStacking = true
                isLoopkill = false
                loopkillTargetName = nil
                isStomping = false
                stompTargetName = nil
                isBringing = false
                bringTargetName = nil
                bringPhase = nil
                isManualTarget = true
                isGuardActive = false
                currentTarget = cmdTarget
                isOrbiting = true
                isFollowing = false
                isHiding = false
                formationMode = "none"
                isGlueFollowActive = true
                if not currentWeapon then
                    currentWeapon = "Knife"
                    attackSubMode = "knife"
                end
                equipTool(currentWeapon or "Knife")
                status.Text = "Status: STACK " .. cmdTarget.Name
            else
                status.Text = "Stack: player not found!"
            end
        end
    elseif cleanCmd == "stomp" or cleanCmd == "s" then
        cmdTarget = findPlayer(param)
        if cmdTarget and not isWhitelisted(cmdTarget) then
            stompTargetName = cmdTarget.Name
            isStomping = true
            isGlueFollowActive = true
            isLoopkill = false
            loopkillTargetName = nil
            isBringing = false
            bringTargetName = nil
            bringPhase = nil
            isManualTarget = true
            isGuardActive = false
            currentTarget = cmdTarget
            isOrbiting = true
            isFollowing = false
            isHiding = false
            formationMode = "none"
            currentWeapon = "Knife"
            attackSubMode = "knife"
            if shouldVoidHuntTarget(cmdTarget) then
                beginCombatChase(cmdTarget)
            else
                if releaseGlueFollowLock then releaseGlueFollowLock() end
                isGlueFollowActive = true
                if equipTool then pcall(equipTool, "Knife") end
            end
            status.Text = "Status: STOMPING " .. cmdTarget.Name
        end
    elseif cleanCmd == "bring" or cleanCmd == "br" then
        cmdTarget = findPlayer(param)
        if cmdTarget and not isWhitelisted(cmdTarget) then
            bringTargetName = cmdTarget.Name
            isBringing = true
            bringPhase = "knock"
            bringPhaseTick = tick()
            isGlueFollowActive = true
            isStomping = false
            stompTargetName = nil
            isLoopkill = false
            loopkillTargetName = nil
            isStacking = false
            stackTargetName = nil
            isManualTarget = true
            isGuardActive = false
            currentTarget = cmdTarget
            isOrbiting = true
            isFollowing = false
            isHiding = false
            formationMode = "none"
            currentWeapon = "Knife"
            attackSubMode = "knife"
            if equipTool then pcall(equipTool, "Knife") end
            status.Text = "Status: BRING (knock) " .. cmdTarget.Name
        end
    elseif cleanCmd == "knock" or cleanCmd == "k" then
        cmdTarget = findPlayer(param)
        if cmdTarget and not isWhitelisted(cmdTarget) then
            isLoopkill = false
            loopkillTargetName = nil
            isStomping = false
            stompTargetName = nil
            isBringing = false
            bringTargetName = nil
            bringPhase = nil
            isManualTarget = true
            isGuardActive = false
            currentTarget = cmdTarget
            isOrbiting = true
            isFollowing = false
            formationMode = "none"
            beginCombatChase(cmdTarget)
            status.Text = "Status: CHASE K -> " .. cmdTarget.Name
        end
    elseif cleanCmd == "knife" then
        attackSubMode = "knife"
        currentWeapon = "Knife"
        equipTool("Knife")
        status.Text = "Weapon: Knife"
    elseif cleanCmd == "revolver" or cleanCmd == "gun" then
        attackSubMode = "combat"
        currentWeapon = "Revolver"
        equipTool("Revolver")
        status.Text = "Weapon: Revolver"
    elseif cleanCmd == "box" then
        formationMode = "box"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Box"
    elseif cleanCmd == "triangle" then
        formationMode = "triangle"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Triangle"
    elseif cleanCmd == "wedge" or cleanCmd == "v" then
        formationMode = "wedge"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Wedge"
    elseif cleanCmd == "circle" then
        formationMode = "circle"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Circle"
    elseif cleanCmd == "line" then
        formationMode = "line"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        status.Text = "Formation: Line"
    elseif cleanCmd == "shield" then
        formationMode = "shield"
        isFollowing = true; isHiding = false; isOrbiting = false; isLoopkill = false; isStomping = false; isManualTarget = false; isGuardActive = false
        isGlueFollowActive = true
        status.Text = "Formation: Shield"
    elseif cleanCmd == "sit" then
        formationMode = "sit"
        isFollowing = true
        isHiding = false
        isOrbiting = false
        isLoopkill = false
        isStomping = false
        isBringing = false
        bringTargetName = nil
        bringPhase = nil
        isManualTarget = false
        isGuardActive = false
        if LocalPlayer ~= speaker and isWhitelisted(LocalPlayer) then
            host = getLeader()

                altIndex = 1
                pcall(function()
                    altIndex = select(1, getAltIndex())
                end)
                task.delay((altIndex - 1) * 0.1, function()
                    if isUnloaded then return end
                    fireSitEmote()
                end)
            end
        end
        status.Text = "Formation: Sit"

        stopAndHide()

        startStand()

        if param then
            pLower = string.lower(param)
            if pLower == "above" or pLower == "sky" or pLower == "top" then
                standPositionMode = "above"
            elseif pLower == "ring" or pLower == "default" or pLower == "side" then
                standPositionMode = "default"
            elseif pLower == "hidden" or pLower == "under" then
                standPositionMode = "hidden"
            else
                status.Text = "Pos: above | ring | hidden"
                return
            end
        else
            standPositionMode = (standPositionMode == "above" and "default" or "above")
        end
        status.Text = "Attack position: " .. standPositionMode
    elseif cleanCmd == "above" then
        standPositionMode = "above"
        status.Text = "Attack position: above"
    elseif cleanCmd == "glue" then
        if param then
            pLower = string.lower(param)
            if pLower == "on" then
                toggleGlueFollow(true)
            elseif pLower == "off" then
                toggleGlueFollow(false)
            else
                glueTarget = findPlayer(param)
                if glueTarget == LocalPlayer then
                    toggleGlueFollow()
                elseif not glueTarget then
                    status.Text = "Glue: player not found!"
                end
            end
        else
            toggleGlueFollow()
        end
    elseif cleanCmd == "antistomp" or cleanCmd == "as" then
        if param then
            pLower = string.lower(param)
            if pLower == "on" then
                toggleAntiStomp(true)
            elseif pLower == "off" then
                toggleAntiStomp(false)
            else
                asTarget = findPlayer(param)
                if asTarget == LocalPlayer then
                    toggleAntiStomp()
                elseif not asTarget then
                    status.Text = "Anti stomp: player not found!"
                end
            end
        else
            toggleAntiStomp()
        end
    elseif cleanCmd == "say" or cleanCmd == "chat" then
        sayText = msg:match("^[%./!]*[sS][aA][yY]%s+(.*)$")
            or msg:match("^[%./!]*[cC][hH][aA][tT]%s+(.*)$")
        if not sayText or sayText:gsub("%s", "") == "" then
            status.Text = "Say: usage say hello"
            return
        end
        if LocalPlayer == speaker then
            return
        end
        if not isWhitelisted(LocalPlayer) then
            return
        end
        host = getLeader()
        if host and LocalPlayer == host then
            return
        end
        altIndex = 1
        pcall(function()
            altIndex = select(1, getAltIndex())
        end)
        task.delay((altIndex - 1) * 0.12, function()
            if isUnloaded then return end
            sendAltChat(sayText)
        end)
        status.Text = "Say: " .. string.sub(sayText, 1, 40)
    elseif cleanCmd == "rejoin" or cleanCmd == "rj" then
        TeleportService:Teleport(game.PlaceId, LocalPlayer)
    elseif cleanCmd == "leave" then
        game:Shutdown()
    end

local lastAutoFireTick = 0

RunService.RenderStepped:Connect(function()
    if isUnloaded or resetInProgress then return end
    local char = getChar(LocalPlayer)
    if not char then return end
    local myHrp = char:FindFirstChild("HumanoidRootPart")
    local myHum = char:FindFirstChildOfClass("Humanoid")
    if not myHrp or not myHum then return end

    if isHoldingGun() and myHum.Health > 0 and (not isHiding or isCombatTaskActive()) then
        setRightClick(true)
    else
        setRightClick(false)
    end

    if shouldParkInVoid() then
        applyVoidHideProtection(char)
        applyVoidHideCFrame(myHrp)
        if not shouldRunMainLoopWhileHiding() then
            return
        end
    end

    -- Stomp Mode Check
    if isStomping and stompTargetName then
        local targetPlayer = findPlayer(stompTargetName)
        if targetPlayer then
            local _, targetChar = getPlayerRootPart(targetPlayer)
            if not targetChar then return end
            local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
            local targetBE = targetChar:FindFirstChild("BodyEffects")
            local targetIsKnocked = (targetBE and targetBE:FindFirstChild("K.O") and targetBE["K.O"].Value == true) or (targetHum and targetHum.Health <= 0)
            local normalTorso = targetChar:FindFirstChild("Torso") or targetChar:FindFirstChild("UpperTorso") or getCharRootPart(targetChar)

            if targetIsKnocked and normalTorso then
                isOrbiting = false
                local dead = targetBE and targetBE:FindFirstChild("Dead")
                if dead and dead.Value == true then
                    currentTarget = targetPlayer
                    isOrbiting = true
                    return
                end
                local altIndex, totalAlts = getAltIndex()
                local stompCF = getGlueStompGoalCFrame(normalTorso, altIndex, totalAlts)
                if shouldVoidHuntTarget(targetPlayer) then
                    applyVoidHideProtection(char)
                    glueSpot = nil
                    local pin = normalTorso.AssemblyRootPart or normalTorso
                    snapVoidHuntCFrame(myHrp, stompCF, pin)
                elseif isGlueFollowActive then
                    glueSpot = stompCF
                else
                    myHrp.AssemblyLinearVelocity = Vector3.zero
                    myHrp.AssemblyAngularVelocity = Vector3.zero
                    myHrp.CFrame = stompCF
                end
                local now = tick()
                if now - lastStompETick >= 0.35 then
                    lastStompETick = now
                    pcall(function()
                        local mainEvent = ReplicatedStorage:FindFirstChild("MainEvent")
                        if mainEvent then
                            pcall(function() mainEvent:FireServer("Stomp") end)
                        end
                        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.E, false, game)
                        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.E, false, game)
                    end)
                end
                return
            else
                currentTarget = targetPlayer
                isOrbiting = true
                isGlueFollowActive = true
            end
        else
            isStomping = false
            stompTargetName = nil
            currentTarget = nil
            isFollowing = true
            status.Text = "Status: FOLLOWING"
        end
    end

    -- Bring Mode (knock -> grab -> host -> drop)
    if isBringing and bringTargetName then
        local targetPlayer = findPlayer(bringTargetName)
        if not targetPlayer then
            finishBringFollow()
        else
            local targetChar = getChar(targetPlayer) or targetPlayer.Character
            if not targetChar then return end
            local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
            local targetBE = targetChar:FindFirstChild("BodyEffects")
            local targetIsKnocked = (targetBE and targetBE:FindFirstChild("K.O") and targetBE["K.O"].Value == true)
                or (targetHum and targetHum.Health <= 0)
            local normalTorso = targetChar:FindFirstChild("UpperTorso")
                or targetChar:FindFirstChild("Torso")
                or getCharRootPart(targetChar)
            local altIndex, totalAlts = getAltIndex()
            local isGrabber = (altIndex == 1)

            if not targetIsKnocked or not normalTorso then
                bringPhase = "knock"
                currentTarget = targetPlayer
                isOrbiting = true
                isGlueFollowActive = true
                if currentWeapon then
                    equipTool(currentWeapon)
                else
                    currentWeapon = "Knife"
                    attackSubMode = "knife"
                    equipTool("Knife")
                end
            else
                isOrbiting = false
                if bringPhase == "knock" then
                    bringPhase = "hover"
                    bringPhaseTick = tick()
                    pcall(unequipAll)
                    status.Text = "Status: BRING (grab) " .. targetPlayer.Name
                end

                if bringPhase == "hover" then
                    if not isGrabber then
                        local hostGoal = getHostTeleportCFrame()
                        if hostGoal then
                            myHrp.AssemblyLinearVelocity = Vector3.zero
                            myHrp.CFrame = hostGoal
                        end
                        pcall(unequipAll)
                        if tick() - bringPhaseTick >= 1.0 then
                            finishBringFollow()
                        end
                        return
                    end

                    local hoverCF = getGlueStompGoalCFrame(normalTorso, 1, 1)
                    glueSpot = nil
                    clearGluePhysics(myHrp)
                    myHrp.AssemblyLinearVelocity = Vector3.zero
                    myHrp.AssemblyAngularVelocity = Vector3.zero
                    myHrp.CFrame = hoverCF

                    if isLocalGrabbing() then
                        bringPhase = "return"
                        bringPhaseTick = tick()
                        status.Text = "Status: BRING (to host)"
                    else
                        fireGrabbing()
                        if tick() - bringPhaseTick >= C.BRING_HOVER_SEC then
                            status.Text = "Status: BRING (grab fail)"
                            finishBringFollow()
                        end
                    end
                    return
                elseif bringPhase == "return" then
                    if not isGrabber then
                        finishBringFollow()
                        return
                    end
                    glueSpot = nil
                    clearGluePhysics(myHrp)
                    pcall(unequipAll)
                    teleportAltToHost(myHrp)
                    if tick() - bringPhaseTick >= 0.55 then
                        bringPhase = "drop_wait"
                        bringPhaseTick = tick()
                        status.Text = "Status: BRING (drop)"
                    end
                    return
                elseif bringPhase == "drop_wait" then
                    pcall(unequipAll)
                    if isGrabber then
                        teleportAltToHost(myHrp)
                        if tick() - bringPhaseTick >= C.BRING_DROP_SEC then
                            fireDropGrabbing()
                            task.delay(0.12, function()
                                pcall(fireDropGrabbing)
                            end)
                            finishBringFollow()
                        end
                    else
                        finishBringFollow()
                    end
                    return
                end
            end
        end
    end

    -- Guard / GKILL Mode
    local leader = getLeader()
    local leaderHrp = getLeaderHrp()

    if isGuardActive and not isManualTarget and not isLoopkill and not isStacking and not isStomping and not isBringing and tick() >= bringCombatLockUntil then
        if leaderHrp then
            local nearestEnemy = nil
            local voidHider = nil
            local voidHiderScore = -1
            local shortestDist = 50.0
            local players = Players:GetPlayers()
            for i = 1, #players do
                local p = players[i]
                if p ~= leader and p ~= LocalPlayer and not isOwnerPlayer(p) and not isWhitelisted(p) then
                    local tHrp, pChar = getPlayerRootPart(p)
                    if not pChar then continue end
                    local tHum = pChar:FindFirstChildOfClass("Humanoid")
                    if tHum and (tHum.Health <= 0 or isCharacterKnocked(pChar)) then continue end
                    if isVoidHidingPlayer(p) then
                        local pos = tHrp and tHrp.Position or getPartWorldPosition(pChar)
                        local score = pos and (pos.Magnitude + getTargetWorldDistance(p)) or getTargetWorldDistance(p)
                        if score > voidHiderScore then
                            voidHiderScore = score
                            voidHider = p
                        end
                    end
                    if tHrp then
                        local dist = (tHrp.Position - leaderHrp.Position).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            nearestEnemy = p
                        end
                    end
                end
            end
            if voidHider then
                nearestEnemy = voidHider
            end
            if nearestEnemy then
                local newGuardTarget = currentTarget ~= nearestEnemy
                currentTarget = nearestEnemy
                isOrbiting = true
                isFollowing = false
                formationMode = "none"
                isGlueFollowActive = true
                if newGuardTarget and not currentWeapon then
                    currentWeapon = "Knife"
                    attackSubMode = "knife"
                end
                if newGuardTarget and shouldVoidHuntTarget(nearestEnemy) then
                    pcall(beginVoidHideEngage, nearestEnemy)
                    if releaseGlueFollowLock then releaseGlueFollowLock() end
                    status.Text = "Status: VOID HUNT (GK) -> " .. nearestEnemy.Name
                    status.TextColor3 = Color3.fromRGB(255, 80, 120)
                elseif isHiding and newGuardTarget then
                    pcall(beginVoidHideEngage, nearestEnemy)
                    status.Text = "Status: VOID GUARD -> " .. nearestEnemy.Name
                    status.TextColor3 = Color3.fromRGB(255, 120, 80)
                elseif newGuardTarget then
                    status.Text = "Status: GK -> " .. nearestEnemy.Name
                    status.TextColor3 = Color3.fromRGB(100, 200, 255)
                end
            elseif isOrbiting and currentTarget and not isManualTarget then
                returnToVoidHideAfterGuard()
            end
        end
    end

    if isLoopkill and loopkillTargetName then
        currentTarget = findPlayer(loopkillTargetName)
    end

    if isStacking and stackTargetName then
        local stackPlayer = findPlayer(stackTargetName)
        if stackPlayer and not isWhitelisted(stackPlayer) then
            currentTarget = stackPlayer
            isOrbiting = true
            isFollowing = false
        end
    end

    -- Combat & Orbiting
    local isKnifeMode = false
    if tick() < bringCombatLockUntil then
        pcall(unequipAll)
        isOrbiting = false
        currentTarget = nil
        isManualTarget = false
    end

    if isOrbiting and currentTarget and tick() >= bringCombatLockUntil then
        if not isStomping and not isBringing then
            isGlueFollowActive = true
        end
        if isWhitelisted(currentTarget) then
            if not isLoopkill and not isManualTarget then
                if isGuardActive then
                    returnToVoidHideAfterGuard()
                else
                    currentTarget = nil
                    isOrbiting = false
                    isFollowing = true
                end
            else
                fullStopAndReturn()
            end
            return
        end

        local targetChar = getChar(currentTarget) or currentTarget.Character
        local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")
        local targetHrp = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
        local orbitPart = getOrbitPart(targetChar)

        if isGuardActive and not isManualTarget and targetChar and isCharacterKnocked(targetChar) then
            returnToVoidHideAfterGuard()
            return
        end

        if not targetChar or not targetHum or targetHum.Health <= 0 or not targetHrp or not orbitPart then
            if isLoopkill or isStacking then
                return
            elseif not isManualTarget and not isGuardActive and not isStomping and not isBringing then
                currentTarget = nil
                isOrbiting = false
                isFollowing = true
                return
            elseif isGuardActive then
                returnToVoidHideAfterGuard()
                return
            elseif not isStomping and not isBringing then
                fullStopAndReturn()
                return
            end
        end

        if isBringing and bringPhase ~= "knock" then
            return
        end

        if currentWeapon then
            equipTool(currentWeapon)
        elseif shouldUseInfiniteChase(currentTarget) or isVoidHideEngaging() then
            if currentWeapon ~= "Knife" or attackSubMode ~= "knife" then
                currentWeapon = "Knife"
                attackSubMode = "knife"
            end
            equipTool("Knife")
        end

        local activeTool = char:FindFirstChildOfClass("Tool")
        local voidHuntCombat = shouldUseInfiniteChase(currentTarget)
        local aimPart = voidHuntCombat and getVoidHuntAimPart(targetChar)
            or getAimPart(targetChar)
            or targetChar:FindFirstChild("Head")
            or targetHrp
        if aimPart then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, aimPart.Position)
        end

        if activeTool and not isBringing then
            local tName = string.lower(activeTool.Name)
            if tName:find("knife") or tName:find("pitchfork") or tName:find("stopsign") or tName:find("whip") then
                isKnifeMode = true
                pcall(function() activeTool:Activate() end)
                if voidHuntCombat then
                    pcall(function() activeTool:Activate() end)
                end
            else
                isKnifeMode = false
                local now = tick()
                local voidHuntFire = voidHuntCombat
                local fireCd = voidHuntFire and C.VOID_HUNT_FIRE_COOLDOWN or 0.13
                if aimPart and now - lastAutoFireTick >= fireCd then
                    lastAutoFireTick = now
                    pcall(function() activeTool:Activate() end)
                    local mainEvent = ReplicatedStorage:FindFirstChild("MainEvent")
                    if mainEvent and activeTool:FindFirstChild("Handle") then
                        local shotPos = aimPart.Position
                        local handlePos = activeTool.Handle.CFrame.Position
                        mainEvent:FireServer("UpdateMousePos", shotPos)
                        mainEvent:FireServer("UpdateMousePosI", shotPos)
                        mainEvent:FireServer(
                            "ShootGun",
                            activeTool.Handle,
                            handlePos,
                            shotPos,
                            aimPart,
                            Vector3.new(0, 0, -1)
                        )
                        if voidHuntFire then
                            local shotDir = shotPos - handlePos
                            if shotDir.Magnitude > 1e-3 then
                                shotDir = shotDir.Unit
                            else
                                shotDir = Vector3.new(0, 0, -1)
                            end
                            mainEvent:FireServer("UpdateMousePos", shotPos)
                            mainEvent:FireServer(
                                "ShootGun",
                                activeTool.Handle,
                                handlePos,
                                shotPos,
                                aimPart,
                                shotDir
                            )
                        end
                    end
                    checkAutoReloadOnTool(activeTool)
                end
            end
        end

        local altIndex, totalAlts = getAltIndex()

        local goalCF = buildCombatOrbitCFrame(orbitPart, altIndex, totalAlts)
        local voidSnap = shouldVoidHuntTarget(currentTarget)

        if voidSnap then
            applyVoidHideProtection(char)
            glueSpot = nil
            local pinTorso = targetChar:FindFirstChild("UpperTorso") or targetChar:FindFirstChild("Torso")
            local pin = (pinTorso and (pinTorso.AssemblyRootPart or pinTorso))
                or (targetHrp and (targetHrp.AssemblyRootPart or targetHrp))
                or orbitPart
            if goalCF then snapVoidHuntCFrame(myHrp, goalCF, pin) end
        elseif not isGlueFollowActive or not glueSpot then
            myHrp.AssemblyLinearVelocity = Vector3.zero
            myHrp.CFrame = goalCF
        end
    else
        isKnifeMode = false
        if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) and SilentAim.Enabled then
            local best = getBestSilentTarget()
            if best and best.Part then
                Camera.CFrame = CFrame.new(Camera.CFrame.Position, best.Part.Position)
            end
        end
    end

    -- Hitbox Expansion
    local allPlayers = Players:GetPlayers()
    for i = 1, #allPlayers do
        local otherPlayer = allPlayers[i]
        if otherPlayer ~= LocalPlayer then
            local oChar = getChar(otherPlayer)
            local targetRoot = oChar and oChar:FindFirstChild("HumanoidRootPart")
            local targetHum = oChar and oChar:FindFirstChildOfClass("Humanoid")
            if targetRoot and targetHum then
                local voidHuntMark = (currentTarget == otherPlayer and isOrbiting and shouldUseInfiniteChase(otherPlayer))
                    or (isOrbiting and currentTarget == otherPlayer and isVoidHidingPlayer(otherPlayer))
                if voidHuntMark and targetHum.Health > 0 and not isWhitelisted(otherPlayer) then
                    local rootPos = targetRoot.Position
                    local voidScale = math.max(math.abs(rootPos.X), math.abs(rootPos.Y), math.abs(rootPos.Z), rootPos.Magnitude, 1)
                    local hb = voidScale >= C.VOID_HUNT_DEEP_STUDS and C.VOID_HUNT_HITBOX_DEEP or C.VOID_HUNT_HITBOX
                    targetRoot.Size = Vector3.new(hb, hb, hb)
                    targetRoot.Transparency = 0.65
                    targetRoot.CanCollide = false
                elseif isKnifeMode and targetHum.Health > 0 and not isWhitelisted(otherPlayer) then
                    targetRoot.Size = Vector3.new(C.hitboxScaleSize, C.hitboxScaleSize, C.hitboxScaleSize)
                    targetRoot.Transparency = 0.5
                    targetRoot.CanCollide = false
                else
                    targetRoot.Size = C.defaultHitboxSize
                    targetRoot.Transparency = 1
                end
            end
        end
    end

    if not isFollowing or shouldParkInVoid() or isOrbiting then return end
    if isGlueFollowActive and glueSpot then return end

    -- Follow Formations (alts ring around host; host account does not take a slot)
    local host = getLeader()
    if leaderHrp and host and LocalPlayer ~= host then
        local altIndex, totalAlts = getAltIndex()
        applyFormationFollow(myHrp, leaderHrp, formationMode, altIndex, totalAlts)
    end
end)

;(function()
    local lastChatDedupeKey = ""
    local lastChatDedupeTick = 0

    local function handleChat(speaker, message)
        if isUnloaded or not speaker or not message then return end
        local dedupeKey = tostring(speaker.UserId) .. "\31" .. message
        local now = tick()
        if dedupeKey == lastChatDedupeKey and (now - lastChatDedupeTick) < 0.75 then
            return
        end
        lastChatDedupeKey = dedupeKey
        lastChatDedupeTick = now
        executeCommand(speaker, message)
    end

    local chatPlayers = Players:GetPlayers()
    for i = 1, #chatPlayers do
        local p = chatPlayers[i]
        p.Chatted:Connect(function(msg) handleChat(p, msg) end)
    end
    Players.PlayerAdded:Connect(function(p)
        p.Chatted:Connect(function(msg) handleChat(p, msg) end)
    end)

    local function hookTextChannel(channel)
        if not channel or not channel:IsA("TextChannel") then return end
        channel.MessageReceived:Connect(function(txt)
            local text = txt and txt.Text
            if not text or text == "" then return end
            local sender = txt.TextSource and resolveChatSender(txt.TextSource)
            if sender then handleChat(sender, text) end
        end)
    end

    pcall(function()
        for _, channel in ipairs(TextChatService:GetTextChannels()) do
            hookTextChannel(channel)
        end
        TextChatService.ChildAdded:Connect(function(child)
            hookTextChannel(child)
        end)
    end)

    task.spawn(function()
        local channels = TextChatService:WaitForChild("TextChannels", 25)
        if not channels then return end
        local general = channels:WaitForChild("RBXGeneral", 25)
        if general then hookTextChannel(general) end
    end)

    pcall(function()
        local legacy = ReplicatedStorage:WaitForChild("DefaultChatSystemChatEvents", 10)
        local onMsg = legacy and legacy:WaitForChild("OnMessageDoneFiltering", 10)
        if onMsg then
            onMsg.OnClientEvent:Connect(function(packet)
                if type(packet) ~= "table" then return end
                local uid = packet.FromSpeakerUserId or packet.SpeakerUserId or 0
                local sender = uid > 0 and Players:GetPlayerByUserId(uid) or nil
                local text = packet.Message or packet.msg
                if sender and text and text ~= "" then handleChat(sender, text) end
            end)
        end
    end)
end)()

    isAntiStompActive = true
    local c = getChar(LocalPlayer)
    if c and armAntiStomp then
        task.spawn(function()
            armAntiStomp(c)
        end)
    end
end)
